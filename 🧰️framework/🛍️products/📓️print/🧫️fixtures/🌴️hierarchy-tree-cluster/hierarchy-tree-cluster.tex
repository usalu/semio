\ExplSyntaxOn
% 📤 One record per node field, in the hierarchy's own preorder (d3 eachBefore).
\cs_new_protected:Npn \VizProbeField #1#2#3 {
  \tl_clear:N \l_tmpa_tl
  \semio_viz_hierarchy_map_pre:nn {#2} {
    \tl_put_right:Nx \l_tmpa_tl
      { \tl_if_empty:NF \l_tmpa_tl { , } \semio_viz_hierarchy_get:nnn {#2} {#3} {##1} }
  }
  \semio_viz_probe_values:nV {#1} \l_tmpa_tl
}
\ExplSyntaxOff
