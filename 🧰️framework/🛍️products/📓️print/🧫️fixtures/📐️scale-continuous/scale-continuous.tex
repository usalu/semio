% 🧪 Probe fixture of case scale-continuous, every scenario.
% One document per scale kind: the same scale is applied, inverted, ticked and niced, and each of
% the four answers is written under its own key so a wrong transform is a named failure rather
% than a single opaque mismatch.
\documentclass{article}
\usepackage{semio-viz-scale}
\usepackage{semio-viz-probe}
\ExplSyntaxOn
\fp_new:N \l_probe_value_fp
\seq_new:N \l_probe_seq

% 📐 Writes the mapped values of one scale over a list of inputs.
\cs_new_protected:Npn \probemap #1#2#3 {
  \seq_clear:N \l_probe_seq
  \clist_map_inline:nn {#3}
    {
      \semio_viz_scale_map:nnN {#2} {##1} \l_probe_value_fp
      \seq_put_right:Nx \l_probe_seq { \fp_to_decimal:n { round( \l_probe_value_fp , 6 ) } }
    }
  \semio_viz_probe_values:nx {#1} { \seq_use:Nn \l_probe_seq { , } }
}

% ↩️ Writes the inverted values of one scale over a list of range positions.
\cs_new_protected:Npn \probeinvert #1#2#3 {
  \seq_clear:N \l_probe_seq
  \clist_map_inline:nn {#3}
    {
      \semio_viz_scale_invert:nnN {#2} {##1} \l_probe_value_fp
      \seq_put_right:Nx \l_probe_seq { \fp_to_decimal:n { round( \l_probe_value_fp , 6 ) } }
    }
  \semio_viz_probe_values:nx {#1} { \seq_use:Nn \l_probe_seq { , } }
}

% 📏 Writes the tick values of one scale.
\cs_new_protected:Npn \probeticks #1#2#3 {
  \semio_viz_scale_ticks_count:nnN {#2} {#3} \l_probe_seq
  \semio_viz_probe_values:nx {#1} { \seq_use:Nn \l_probe_seq { , } }
}

% 🧹 Writes the domain of one scale, after any `nice` it was declared with.
\cs_new_protected:Npn \probedomain #1#2 {
  \semio_viz_probe_values:nx {#1}
    { \clist_use:cn { g_semio_viz_scale_ \tl_to_str:n {#2} _domain_clist } { , } }
}
\ExplSyntaxOff
