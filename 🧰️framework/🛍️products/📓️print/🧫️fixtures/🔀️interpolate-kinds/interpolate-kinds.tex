% 🧪 Probe fixture of case interpolate-kinds, every scenario.
% `\SemioVizInterpolate` is the public form of the kernel's interpolators, so what it must be held
% to is that every kind it names dispatches onto the interpolator it promises. The command itself
% typesets and is therefore not expandable; the probe calls the protected form behind it and emits
% the token list it fills, which is the same value the document would have set.
% Colours and comma lists travel as strings prefixed with a vertical bar so no digit is lost to the
% probe protocol's numeric classification.
\documentclass{article}
\usepackage{semio-viz-scale}
\usepackage{semio-viz-probe}
\ExplSyntaxOn
\tl_new:N \l_probe_result_tl

% 🔀 Writes one interpolated colour of the kind #2 between #3 and #4 at #5, as a string.
\cs_new_protected:Npn \probeinterpolate #1#2#3#4#5 {
  \semio_viz_interpolate:nnnnN {#2} {#3} {#4} {#5} \l_probe_result_tl
  \semio_viz_probe_values:nx {#1} { { | \l_probe_result_tl } }
}

% 🔢 Writes one interpolated number or list of numbers, so the tolerance profile applies to them.
\cs_new_protected:Npn \probeinterpolatenum #1#2#3#4#5 {
  \semio_viz_interpolate:nnnnN {#2} {#3} {#4} {#5} \l_probe_result_tl
  \semio_viz_probe_values:nx {#1} { \l_probe_result_tl }
}
\ExplSyntaxOff
