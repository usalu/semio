\documentclass{article}
\usepackage{semio-viz-probe}
\usepackage{semio-viz-shape}
\ExplSyntaxOn
% 🥧 Generates one arc and records the SVG path and the centroid it produced.
\cs_new_protected:Npn \semio_probe_arc:nn #1#2 {
  \SemioVizArc[ draw=false, #2 ]
  \exp_args:NnV \SemioVizProbeValues { arc/#1 } \g_semio_viz_shape_result_tl
}
% 🎯 Records only the centroid, which d3 computes without touching the path.
\cs_new_protected:Npn \semio_probe_centroid:nn #1#2 {
  \SemioVizArc[ draw=false, #2 ]
  \exp_args:NnV \SemioVizProbeValues { centroid/#1 } \g_semio_viz_shape_centroid_tl
}
% 🥧 Records the start and then the end angle of every datum, in the data's own order.
\cs_new_protected:Npn \semio_probe_pie:nn #1#2 {
  \SemioVizPie[ #2 ]
  \exp_args:Nnx \SemioVizProbeValues { pie/#1 }
    { \seq_use:Nn \g_semio_viz_pie_start_seq { , } , \seq_use:Nn \g_semio_viz_pie_end_seq { , } }
}
