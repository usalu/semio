% 🧪 Probe fixture of case transform-stack, every scenario.
% The stack transform reads a long-form table `t, series, value` and writes `x, series, y0, y1`.
% Every scenario probes the two baselines of every series separately, so a wrong order and a wrong
% offset are different failures.
\documentclass{article}
\usepackage{semio-viz-transform}
\usepackage{semio-viz-probe}
\ExplSyntaxOn
\seq_new:N \l_probe_seq
\tl_new:N \l_probe_cell_tl

% 🥞 Writes one column of a stacked table filtered to one series.
\cs_new_protected:Npn \probestack #1#2#3#4 {
  \seq_clear:N \l_probe_seq
  \seq_map_inline:cn { g_semio_viz_table_ \tl_to_str:n {#2} _rows_seq }
    {
      \probecellof:nnN {##1} { series } \l_probe_cell_tl
      \str_if_eq:VnT \l_probe_cell_tl {#3}
        {
          \probecellof:nnN {##1} {#4} \l_probe_cell_tl
          \seq_put_right:NV \l_probe_seq \l_probe_cell_tl
        }
    }
  \semio_viz_probe_values:nx {#1} { \seq_use:Nn \l_probe_seq { , } }
}

% 🧾 Reads a cell of a stored row by column name of the given table.
\cs_new_protected:Npn \probecellof:nnN #1#2#3 {
  \semio_viz_tr_get:nnN {#1} {#2} #3
}

% 🥞 Writes the series order a stacked table came out in.
\cs_new_protected:Npn \probeorder #1#2 {
  \seq_clear:N \l_probe_seq
  \seq_map_inline:cn { g_semio_viz_table_ \tl_to_str:n {#2} _rows_seq }
    {
      \probecellof:nnN {##1} { series } \l_probe_cell_tl
      \seq_if_in:NVF \l_probe_seq \l_probe_cell_tl
        { \seq_put_right:NV \l_probe_seq \l_probe_cell_tl }
    }
  \semio_viz_probe_values:nx {#1} { \seq_use:Nn \l_probe_seq { , } }
}
