% 🧪 One catalogue kind of showcase family #1: its own probe scenario, then the family itself run
% with nothing but the variant the catalogue entry carries.
\cs_new_protected:Npn \semio_probe_showcase:nn #1#2 {
  \semio_viz_probe_begin:nn { showcase-families } { #1/#2 }
  \semio_viz_family_run:nn {#1} { variant = #2 }
}
% 🧪 Every kind of showcase family #1, from the comma list #2 of its catalogue variants.
\cs_new_protected:Npn \semio_probe_showcase_family:nn #1#2 {
  \clist_map_inline:nn {#2} { \semio_probe_showcase:nn {#1} {##1} }
}
% 🧪 One kind whose catalogue entry names a table of its own: variant #2 of family #1 on table #3,
% exactly the option list `semio-viz-catalog.sty` hands the family.
\cs_new_protected:Npn \semio_probe_showcase_on:nnn #1#2#3 {
  \semio_viz_probe_begin:nn { showcase-families } { #1/#2 }
  \semio_viz_family_run:nn {#1} { data = #3 , variant = #2 }
}
