% 🧪 Probe fixture of case transform-statistics, every scenario.
% Aggregates, quantiles, the kernel density estimate and the five least-squares fits, each written
% out under its own key. The regression scenarios use a small point set whose coefficients are all
% well conditioned, so the comparison measures the fit and not the numerics of a near-singular
% normal-equation matrix.
\documentclass{article}
\usepackage{semio-viz-transform}
\usepackage{semio-viz-probe}
\ExplSyntaxOn
\seq_new:N \l_probe_seq
\tl_new:N \l_probe_cell_tl

% 📊 Writes one column of a derived table.
\cs_new_protected:Npn \probecolumn #1#2#3 {
  \seq_clear:N \l_probe_seq
  \semio_viz_tr_load:n {#2}
  \seq_map_inline:cn { g_semio_viz_table_ \tl_to_str:n {#2} _rows_seq }
    {
      \semio_viz_tr_get:nnN {##1} {#3} \l_probe_cell_tl
      \seq_put_right:NV \l_probe_seq \l_probe_cell_tl
    }
  \semio_viz_probe_values:nx {#1} { \seq_use:Nn \l_probe_seq { , } }
}
\ExplSyntaxOff
\begin{document}
