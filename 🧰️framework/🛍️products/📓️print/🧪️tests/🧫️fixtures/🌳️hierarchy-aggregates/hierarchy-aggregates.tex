% 📤 One record per node field, in the hierarchy's own preorder (d3 eachBefore).
\cs_new_protected:Npn \VizProbeField #1#2#3 {
  \tl_clear:N \l_tmpa_tl
  \semio_viz_hierarchy_map_pre:nn {#2} {
    \tl_put_right:Nx \l_tmpa_tl
      { \tl_if_empty:NF \l_tmpa_tl { , } \semio_viz_hierarchy_get:nnn {#2} {#3} {##1} }
  }
  \semio_viz_probe_values:nV {#1} \l_tmpa_tl
}
% 📥 The same field in the postorder (d3 eachAfter).
\cs_new_protected:Npn \VizProbeFieldPost #1#2#3 {
  \tl_clear:N \l_tmpa_tl
  \semio_viz_hierarchy_map_post:nn {#2} {
    \tl_put_right:Nx \l_tmpa_tl
      { \tl_if_empty:NF \l_tmpa_tl { , } \semio_viz_hierarchy_get:nnn {#2} {#3} {##1} }
  }
  \semio_viz_probe_values:nV {#1} \l_tmpa_tl
}
% 🍃 The identifiers of every leaf, in preorder.
\cs_new_protected:Npn \VizProbeLeaves #1#2 {
  \seq_clear:N \l_tmpa_seq
  \semio_viz_hierarchy_leaves:nN {#2} \l_tmpb_seq
  \seq_map_inline:Nn \l_tmpb_seq
    { \seq_put_right:Nx \l_tmpa_seq { \semio_viz_hierarchy_get:nnn {#2} { id } {##1} } }
  \semio_viz_probe_values:nx {#1} { \seq_use:Nn \l_tmpa_seq { , } }
}
% 🧬 The identifiers on the path from a node up to the root.
\cs_new_protected:Npn \VizProbeAncestors #1#2#3 {
  \seq_clear:N \l_tmpa_seq
  \semio_viz_hierarchy_ancestors:nnN {#2} {#3} \l_tmpb_seq
  \seq_map_inline:Nn \l_tmpb_seq
    { \seq_put_right:Nx \l_tmpa_seq { \semio_viz_hierarchy_get:nnn {#2} { id } {##1} } }
  \semio_viz_probe_values:nx {#1} { \seq_use:Nn \l_tmpa_seq { , } }
}
% 🔗 Every parent/child pair, source then target, in preorder.
\cs_new_protected:Npn \VizProbeLinks #1#2 {
  \seq_clear:N \l_tmpa_seq
  \semio_viz_hierarchy_map_links:nn {#2} {
    \seq_put_right:Nx \l_tmpa_seq
      { \semio_viz_hierarchy_get:nnn {#2} { id } { \l_semio_viz_hierarchy_source_int } }
    \seq_put_right:Nx \l_tmpa_seq
      { \semio_viz_hierarchy_get:nnn {#2} { id } { \l_semio_viz_hierarchy_target_int } }
  }
  \semio_viz_probe_values:nx {#1} { \seq_use:Nn \l_tmpa_seq { , } }
}
