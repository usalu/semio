\documentclass{article}
\usepackage{semio-viz-spatial}
\ExplSyntaxOn
% 📤 Emits one record per clipped cell, the vertices sorted so the starting vertex and winding do not matter.
\cs_new_protected:Npn \semio_probe_cell:nn #1#2
  {
    \seq_set_split:Nnn \l_tmpa_seq { ; } {#2}
    \seq_sort:Nn \l_tmpa_seq
      {
        \fp_compare:nNnTF { \clist_item:nn {##1} { 1 } } < { \clist_item:nn {##2} { 1 } }
          { \sort_return_same: }
          {
            \bool_lazy_and:nnTF
              { \fp_compare_p:nNn { \clist_item:nn {##1} { 1 } } = { \clist_item:nn {##2} { 1 } } }
              { \fp_compare_p:nNn { \clist_item:nn {##1} { 2 } } < { \clist_item:nn {##2} { 2 } } }
              { \sort_return_same: }
              { \sort_return_swapped: }
          }
      }
    \use:x
      {
        \exp_not:N \SemioVizProbeEval { spatial/voronoi }
          { \seq_use:Nn \l_tmpa_seq { , } }
      }
  }
\ExplSyntaxOff
