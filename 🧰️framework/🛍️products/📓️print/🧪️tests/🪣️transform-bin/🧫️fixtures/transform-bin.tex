% 🧪 Probe fixture of case transform-bin, every scenario.
% The bin transform reads one numeric column and writes `x0`, `x1`, `count`. All three are probed
% as separate keys, because a histogram with the right counts on the wrong edges is still wrong.
\documentclass{article}
\usepackage{semio-viz-transform}
\usepackage{semio-viz-probe}
\ExplSyntaxOn
\seq_new:N \l_probe_seq
\tl_new:N \l_probe_cell_tl

% 🪣 Writes one column of a binned table.
\cs_new_protected:Npn \probebin #1#2#3 {
  \seq_clear:N \l_probe_seq
  \semio_viz_tr_load:n {#2}
  \seq_map_inline:cn { g_semio_viz_table_ \tl_to_str:n {#2} _rows_seq }
    {
      \semio_viz_tr_get:nnN {##1} {#3} \l_probe_cell_tl
      \seq_put_right:NV \l_probe_seq \l_probe_cell_tl
    }
  \semio_viz_probe_values:nx {#1} { \seq_use:Nn \l_probe_seq { , } }
}
\ExplSyntaxOff
\begin{document}

\SemioVizProbeBegin{transform-bin}{sturges}
\SemioVizTransform{b1}{demo-distribution}[bin=value]
\probebin{bin/sturges/x0}{b1}{x0}
\probebin{bin/sturges/x1}{b1}{x1}
\probebin{bin/sturges/count}{b1}{count}

\SemioVizProbeBegin{transform-bin}{count-five}
\SemioVizTransform{b2}{demo-distribution}[bin=value,thresholds=count:5]
\probebin{bin/five/x0}{b2}{x0}
\probebin{bin/five/x1}{b2}{x1}
\probebin{bin/five/count}{b2}{count}

\SemioVizProbeBegin{transform-bin}{count-twenty}
\SemioVizTransform{b3}{demo-distribution}[bin=value,thresholds=count:20]
\probebin{bin/twenty/x0}{b3}{x0}
\probebin{bin/twenty/x1}{b3}{x1}
\probebin{bin/twenty/count}{b3}{count}

\SemioVizProbeBegin{transform-bin}{explicit-thresholds}
\SemioVizTransform{b4}{demo-distribution}[bin=value,thresholds=4;6;8]
\probebin{bin/explicit/x0}{b4}{x0}
\probebin{bin/explicit/x1}{b4}{x1}
\probebin{bin/explicit/count}{b4}{count}

\SemioVizProbePage
\end{document}
