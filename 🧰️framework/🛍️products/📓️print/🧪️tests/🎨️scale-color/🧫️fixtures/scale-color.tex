% 🧪 Probe fixture of case scale-color, every scenario.
% A colour scale is two independent pieces: the position `t` a sequential or diverging scale
% computes from its domain, and the colour an interpolator produces at that position. The two are
% probed separately, so a wrong domain and a wrong colour space are different failures.
% Colours travel as strings prefixed with a vertical bar so no hexadecimal digit is lost to the
% probe protocol's numeric classification.
\documentclass{article}
\usepackage{semio-viz-scale}
\usepackage{semio-viz-probe}
\ExplSyntaxOn
\fp_new:N \l_probe_value_fp
\seq_new:N \l_probe_seq
\str_new:N \l_probe_hex_str

% 📐 Writes the normalized positions a sequential or diverging scale assigns.
\cs_new_protected:Npn \probepositions #1#2#3 {
  \seq_clear:N \l_probe_seq
  \clist_map_inline:nn {#3}
    {
      \semio_viz_scale_map:nnN {#2} {##1} \l_probe_value_fp
      \seq_put_right:Nx \l_probe_seq { \fp_to_decimal:n { round( \l_probe_value_fp , 6 ) } }
    }
  \semio_viz_probe_values:nx {#1} { \seq_use:Nn \l_probe_seq { , } }
}

% 🌈 Writes one interpolated colour of a two-stop ramp.
\cs_new_protected:Npn \probemix #1#2#3#4#5 {
  \semio_viz_color_interpolate:nnnnN {#2} {#3} {#4} {#5} \l_probe_hex_str
  \semio_viz_probe_values:nx {#1} { { | \l_probe_hex_str } }
}

% 🌈 Writes one sample of a multi-stop ramp.
\cs_new_protected:Npn \proberamp #1#2#3#4 {
  \semio_viz_color_ramp:nnnN {#2} {#3} {#4} \l_probe_hex_str
  \semio_viz_probe_values:nx {#1} { { | \l_probe_hex_str } }
}
\ExplSyntaxOff
