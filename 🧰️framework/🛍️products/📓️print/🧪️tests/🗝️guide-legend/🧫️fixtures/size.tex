\documentclass{article}
\usepackage{semio-viz-guide}

\NewDocumentEnvironment { ProbeFrame } { }
  {
    \bool_set_true:N \l_semio_viz_in_figure_bool
    \fp_set:Nn \l_semio_viz_width_fp { 80 }
    \fp_set:Nn \l_semio_viz_height_fp { 40 }
    \fp_set:Nn \l_semio_viz_pad_fp { 8 }
    \begin{tikzpicture}[ x=1mm , y=1mm ]
  }
  { \end{tikzpicture} \bool_set_false:N \l_semio_viz_in_figure_bool }

\begin{document}
\SemioVizProbeBegin{guide-legend}{size-entries}
\SemioVizProbeOn
\SemioVizScale{radius}{linear}{0,10}{0,5}
\begin{ProbeFrame}
\SemioVizLegend[scale=radius,kind=size,at={10,30},ticks=3]
\end{ProbeFrame}
\end{document}
