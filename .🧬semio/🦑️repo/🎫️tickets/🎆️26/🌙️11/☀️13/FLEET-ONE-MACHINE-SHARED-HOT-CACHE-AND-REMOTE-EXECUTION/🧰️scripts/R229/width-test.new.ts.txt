import{expect,test}from"bun:test";
import{readFileSync}from"node:fs";
import{spawnSync}from"node:child_process";
import{validateJsonSchemaSubset}from"../../../../../🧬️schema/✅️validator/📦️packages/🟦️typescript/🟦️.ts";
import schema from"../🧬️schema/🔣️.json"with{type:"json"};
import{RESTORE_WIDTH_POLICY_V1 as P,restoreWidthV1}from"../🟦️.ts";

/** 🧱️ A build whose units come from a cache may keep more Cargo jobs running than it has compile units, while the compilers stay bounded by the granted units. */
const python=(program:string,input:unknown):any=>{const run=spawnSync("python3",["-I","-c",program],{input:JSON.stringify(input),encoding:"utf8"});if(run.status!==0)throw Error(run.stderr);return JSON.parse(run.stdout);};
const CACHED={RUSTC_WRAPPER:"/w/rustc-wrapper",SEMIO_UNIT_CACHE_ROOT:"/c"};

test("the width equals an independent Python implementation over every granted count, factor and environment",()=>{
 const rows:{granted:number;environment:Record<string,string>}[]=[];
 for(const granted of[0,1,2,3,4,5,8,12,16,40])for(const factor of[undefined,"1","2","3","4","5","0","-1","2.5","x",""])for(const cache of[true,false])rows.push({granted,environment:{...cache?CACHED:{},...factor===undefined?{}:{SEMIO_CARGO_RESTORE_WIDTH:factor}}});
 rows.push({granted:4,environment:{...CACHED,SEMIO_UNIT_CACHE:"off",SEMIO_CARGO_RESTORE_WIDTH:"3"}},{granted:4,environment:{RUSTC_WRAPPER:"/w",SEMIO_CARGO_RESTORE_WIDTH:"3"}},{granted:4,environment:{SEMIO_UNIT_CACHE_ROOT:"/c",SEMIO_CARGO_RESTORE_WIDTH:"3"}});
 const ours=rows.map(row=>restoreWidthV1(row.granted,row.environment));
 const theirs=python(`
import json,sys
P=json.load(sys.stdin)
out=[]
for r in P['rows']:
    g=r['granted'] if isinstance(r['granted'],int) and r['granted']>=1 else 1
    e=r['environment']
    cached=bool(e.get('RUSTC_WRAPPER')) and bool(e.get('SEMIO_UNIT_CACHE_ROOT')) and e.get('SEMIO_UNIT_CACHE')!='off'
    raw=e.get('SEMIO_CARGO_RESTORE_WIDTH')
    try:
        v=float(raw) if raw not in (None,'') else None
    except ValueError:
        v=None
    if raw=='':
        v=0.0
    f=int(v) if cached and v is not None and v==int(v) and v>=1 else None
    factor=min(f,P['maximumFactor']) if f is not None else P['defaultFactor']
    out.append({'jobs':max(g,min(g*factor,P['maximumJobs'])),'slots':g,'factor':factor})
print(json.dumps(out))`,{rows,maximumFactor:P.maximumFactor,defaultFactor:P.defaultFactor,maximumJobs:P.maximumJobs});
 expect(JSON.parse(JSON.stringify(ours))).toEqual(theirs);
 for(const width of ours)expect(validateJsonSchemaSubset(schema.definitions.Width,width,schema)).toEqual([]);
});

test("by default nothing changes, and a cached build widens only when asked, bounded in factor and in jobs, never below the granted units",()=>{
 expect(restoreWidthV1(2,CACHED)).toEqual({jobs:2,slots:2,factor:1});
 expect(restoreWidthV1(2,{...CACHED,SEMIO_CARGO_RESTORE_WIDTH:"3"})).toEqual({jobs:6,slots:2,factor:3});
 expect(restoreWidthV1(2,{...CACHED,SEMIO_CARGO_RESTORE_WIDTH:"9"})).toEqual({jobs:8,slots:2,factor:4});
 expect(restoreWidthV1(10,{...CACHED,SEMIO_CARGO_RESTORE_WIDTH:"4"})).toEqual({jobs:16,slots:10,factor:4});
 expect(restoreWidthV1(20,{...CACHED,SEMIO_CARGO_RESTORE_WIDTH:"4"})).toEqual({jobs:20,slots:20,factor:4});
 expect(restoreWidthV1(2,{SEMIO_CARGO_RESTORE_WIDTH:"3"})).toEqual({jobs:2,slots:2,factor:1});
 expect(P.defaultFactor).toBe(1);
});

test("the Cargo path applies the width to the jobs and the compile slots and writes the width to the ledger and the log (source shape)",()=>{
 const native=readFileSync(new URL("../../🟦️.ts",import.meta.url),"utf8");
 expect(native).toContain("restoreWidthV1(facts.parallelism,environment)");
 expect(native).toContain("environment.CARGO_BUILD_JOBS=String(width.jobs)");
 expect(native).toContain("environment.SEMIO_UNIT_CACHE_SLOTS=String(width.slots)");
 expect(native).toContain('phase:"cargo:jobs"');
 expect(native).toContain("[restore-width]");
});
