RADIUS = 0.14


def pre(a):
    bones = {b['id']: b for b in a['rig']['bones']}
    seat = bones['seat']
    seat.update(angleDeg=90, lengthM=0.15, joint={'type': 'revolute', 'minDeg': 0, 'maxDeg': 0})
    seat['title'] = {'en': 'Swivel seat plate (turns about the vertical seat axis)', 'de': 'Drehbare Sitzplatte (dreht um die senkrechte Sitzachse)'}
    seat['mass'].update(comX=0.01, comY=0.0)
    pad = bones['pad']
    pad.update(parent='frame', x=0, y=0.72)
    b = bones['backrest']
    b.update(x=0.022, y=RADIUS, angleDeg=0, lengthM=0.15, joint={'type': 'fixed'})
    b['title'] = {'en': 'Low backrest on the swivel seat (radius 0.14 m)', 'de': 'Niedrige Lehne am Drehsitz (Radius 0,14 m)'}
    b['mass'].update(comX=0.1, comY=0.0)
    for sc in ('500', '200', '100', '50'):
        for sh in a['representations'][sc]['geometry']['elevation']:
            if sc != '500' and sh['bone'] == 'backrest':
                if sh['id'] == 'back':
                    sh.pop('points', None)
                    sh['points'] = [[-0.025, -0.03], [0.025, -0.03], [0.025, 0.15], [-0.025, 0.15]]
                if sh['id'] == 'back-pad':
                    sh['d'] = 'M -0.02 0.09 L -0.02 0.135 Q -0.02 0.15 -0.005 0.15 L 0.005 0.15 Q 0.02 0.15 0.02 0.135 L 0.02 0.09 Z'
                turn_shape(sh, 0)
            elif sh['bone'] == 'seat':
                if sc == '500':
                    sh['d'] = 'M -0.18 -0.048 L 0.18 -0.048 L 0.18 0.052 L -0.07 0.052 L -0.07 0.172 L -0.18 0.172 Z'
                move_shape(sh, 0, 0.698)
                sh['bone'] = 'frame'
    for p in a['poses']:
        if 'backrest' in p['bones']:
            sign = -1 if p['bones']['backrest']['dx'] < 0 else 1
            del p['bones']['backrest']
            p['bones']['seat'] = {'twistDeg': 90 * sign}
    texts = {
        'turned-90': ('Unoccupied seat turned a quarter turn: the backrest swings round the seat circle from the rear to the side.', 'Unbesetzter Sitz um eine Vierteldrehung gedreht: die Lehne wandert auf dem Sitzkreis vom Heck zur Seite.'),
        'turned-270-key': ('Backrest on the opposite side of the seat circle.', 'Lehne auf der Gegenseite des Sitzkreises.'),
    }
    for p in a['poses']:
        if p['id'] in texts:
            p['description'] = {'en': texts[p['id']][0], 'de': texts[p['id']][1]}
    for c in a['clips']:
        if c['id'] == 'swivel-turn':
            c['title'] = {'en': 'Swivel to both sides', 'de': 'Drehung nach beiden Seiten'}
            c['description'] = {'en': 'The unoccupied seat swings a quarter turn to each side and back in 6 s; the backrest runs round the seat circle.', 'de': 'Der unbesetzte Sitz schwenkt in 6 s eine Vierteldrehung zu jeder Seite und zurück; die Lehne läuft auf dem Sitzkreis.'}
    ass = a['structure']['assumptions']
    ass[0] = {'en': 'The stool is a rigid steel frame on three foot contacts in elevation (the front and rear legs coincide); the seat is a body, the pad sits on a prismatic joint and the backrest turns on a revolute swivel.', 'de': 'Der Hocker ist ein starres Stahlgestell auf drei Fußkontakten in der Ansicht (vordere und hintere Beine fallen zusammen); der Sitz ist ein Körper, das Polster sitzt auf einem Schiebegelenk und die Lehne dreht auf einem Drehgelenk.'}
    ass[1] = {'en': 'The swivel is a rotation about the vertical axis; the round seat plate and pad look the same at every azimuth, so they are drawn on the frame and only the backrest rides on the twisting seat bone (axial twist of the vertical seat bone). It sits at radius 0.14 m and its 0.10 m wide pad turns with it in the front and top view.', 'de': 'Die Drehung ist eine Rotation um die Hochachse; die runde Sitzplatte und das Polster sehen bei jedem Winkel gleich aus, deshalb sind sie am Gestell gezeichnet und nur die Lehne sitzt auf dem sich verdrehenden Sitzknochen (axiale Verdrehung des senkrechten Sitzknochens). Sie sitzt auf dem Radius 0,14 m und ihr 0,10 m breites Polster dreht in Vorder- und Draufsicht mit.'}


def noted(s, sc):
    build(s, sc)
    return NOTES[sc]


if __name__ == '__main__':
    run('bar-stool', noted, pre)
