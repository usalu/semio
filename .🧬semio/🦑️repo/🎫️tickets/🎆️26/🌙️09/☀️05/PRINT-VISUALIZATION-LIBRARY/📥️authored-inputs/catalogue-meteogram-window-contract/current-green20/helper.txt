export async function compileNativeMeteogramWindowControls(workDir:string):Promise<void>{
  const neutral=fixture.nativeMeteogramWindowControls,c=neutral.vectors,validate=new Ajv2020({strict:false}).compile(neutral.schema),before=JSON.stringify(c),failures:string[]=[],cases=c.heights.flatMap(height=>c.windows.map(window=>({...window,height,id:height+"-"+window.id})));
  if(!validate(c))throw Error("Meteogram neutral schema "+JSON.stringify(validate.errors));
  for(const[appearance,language]of[["light","en"],["dark","de"]]as const){
    const directory=join(workDir,appearance),body:{raw:string}[]=[];
    for(const entry of cases){
      const options={mode:"meteogram",panels:c.panels.map(panel=>panel.label+"/"+panel.points.map(point=>point.join(":")).join("+")).join(","),width:c.width,height:entry.height===40?80:40,axes:false,grid:false,domain:entry.domain,range:entry.range},base={chart:{language,theme:{appearance},width:c.width,height:entry.height,margin:{top:0,right:0,bottom:0,left:0},layers:[],tables:[],presets:[{kind:"meteogram",options}]}}as unknown as VizChartSnapshot,original=JSON.stringify(base),changed=changeVizChartValue(base,{path:["presets","0","options","height"],value:entry.height}),replayed=applyVizChartDiff(base,changed.diff),inverse=applyVizChartDiff(replayed.snapshot,inverseVizChartDiff(base,changed.diff));if(changed.messages.length||changed.diff.edits.length!==1||replayed.messages.length||inverse.messages.length||JSON.stringify(base)!==original)throw Error(entry.id+" meteogram mutation/replay/purity refused");deepStrictEqual(inverse.snapshot,base);const inferred=await inferVizChart(replayed.snapshot);if(!inferred.tikz||inferred.diagnostics.some(item=>item.code!=="print.chart.scene-unavailable"))throw Error(entry.id+" meteogram inference "+JSON.stringify(inferred.diagnostics));mkdirSync(join(directory,"emissions"),{recursive:true});await Bun.write(join(directory,"emissions",entry.id+".tex"),inferred.tikz);body.push({raw:"\\clearpage\\SemioVizProbeBegin{meteogram-window-controls}{"+entry.id+"}\\ExplSyntaxOn\\semio_viz_probe_values:nn{window/control}{1}\\ExplSyntaxOff\n"+inferred.tikz.replace(/(\\begin\{VizFigure\}[^\n]*\n)/,"$1\\draw[line width=.01mm] (0,0) rectangle ("+c.width+","+entry.height+");\\special{pdf:literal direct /SemioVizControl"+entry.id+" BMC}\n").replace("\\end{VizFigure}","\\special{pdf:literal direct EMC}\\end{VizFigure}")});
    }
    await compileVizProbeDocument({case:"meteogram-window-controls",scenario:appearance,documentClass:"semio",documentClassOptions:"type=paper,language="+language+",theme="+appearance,geometry:true,packages:["semio-viz"],preamble:["\\title{Meteogram Authored Windows}","\\author{Semio}","\\date{}","\\errorcontextlines=200"],body},{workDir:directory,scenario:undefined,keepWorkDir:true});const pdf=await getDocument({data:new Uint8Array(readFileSync(join(directory,"🧪️probe-out",appearance+".pdf"))),useSystemFonts:true}).promise,paints=new Map<string,ReturnType<typeof nativeControlPdfPaint>>();
    try{for(let page=1;page<=pdf.numPages;page++){const operators=await(await pdf.getPage(page)).getOperatorList();for(const entry of cases){const marker="SemioVizControl"+entry.id;if(operators.argsArray.some((args,index)=>[OPS.beginMarkedContent,OPS.beginMarkedContentProps].includes(operators.fnArray[index]!)&&String(args?.[0]?.name??args?.[0])===marker))paints.set(entry.id,nativeControlPdfPaint(operators,[c.width,entry.height],marker));}}}finally{await pdf.destroy();}await Bun.write(join(directory,"owned-paint.json"),JSON.stringify(Object.fromEntries(paints),null,2));
    for(const entry of cases){try{
      const paint=paints.get(entry.id);if(!paint)throw Error("Meteogram actual marker absent");const height=(entry.height-c.gapMm*(c.panels.length-1))/c.panels.length,curves=paint.paths.filter(path=>!path.closed&&path.points.length===3);if(curves.length!==c.panels.length)throw Error("Meteogram actual panel curve inventory "+curves.length);
      c.panels.forEach((panel,index)=>{const xs=panel.points.map(point=>point[0]!),ys=panel.points.map(point=>point[1]!),extent=(values:number[])=>{const lo=Math.min(...values),hi=Math.max(...values);return[lo-c.padding*(hi-lo),hi+c.padding*(hi-lo)];},domain=entry.domain?entry.domain.split(",").map(Number):extent(xs),range=entry.range?entry.range.split(",").map(Number):extent(ys),x=scaleLinear(domain,[0,c.width]),y=scaleLinear(range,[-index*(height+c.gapMm),height-index*(height+c.gapMm)]),expected=panel.points.map(point=>[x(point[0]!),y(point[1]!)]);expected.forEach((point,vertex)=>equal(curves[index]!.points[vertex]!,point,entry.id+" D3 actual panel "+panel.label+" vertex "+vertex,c.toleranceMm));});
    }catch(error){failures.push(appearance+"/"+entry.id+" "+String(error));}}
  }
  if(JSON.stringify(c)!==before)throw Error("Meteogram neutral fixture changed");if(failures.length)throw Error("Native meteogram window effects:\n"+failures.join("\n"));console.log("[DEBUG] Meteogram 20 actual parent-height/window cases and 120 curve vertices matched independent D3 and canonical replay/inverse/purity");
}
