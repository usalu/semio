  if(process.env.PRINT_NATIVE_GRAMMAR_PHASE==="scientific-controls"){await compileNativeScientificControlEffects(join(workDir,"scientific"));return;}
  if(process.env.PRINT_NATIVE_GRAMMAR_PHASE==="process-controls"){await compileNativeProcessControlEffects(join(workDir,"process"));return;}

/** ⚖️ Checks numeric probes against independent expectations with the protocol's fixed precision. */
function equal(actual: readonly number[], expected: readonly number[], label: string, tolerance = 1e-4): void {
  if (actual.length !== expected.length || actual.some((n, i) => !Number.isFinite(n) || Math.abs(n - expected[i]!) > tolerance)) throw new Error(`${label}: actual=${JSON.stringify(actual)}, expected=${JSON.stringify(expected)}`);
}

              equal(value,oracle.values[index]!,entry.id+" visible PDF shape "+index,.01);visible++;});
          }catch(error){failures.push(theme+"/"+entry.id+": "+String(error));}
        }

      }
    }finally{await pdf.destroy();}
  }

}

type NativeFamilyPdfShape={kind:"circle"|"rect";bounds:number[];paint:number};

/** 🔬️ Measures visible native glyphs relative to an independently declared PDF canvas reference. */
function nativeFamilyPdfShapes(operators:Awaited<ReturnType<PDFPageProxy["getOperatorList"]>>,canvas:number[]):NativeFamilyPdfShape[]{
  const shapes:{kind:"circle"|"rect";points:number[][];matrix:number[];paint:number}[]=[],stack:{matrix:number[];width:number}[]=[];
  const point=scaleLinear([0,1],[0,72/25.4]);
  let matrix=[1,0,0,1,0,0],width=1,reference:number[]|undefined;
  for(const [index,id]of operators.fnArray.entries()){
    const args=operators.argsArray[index];
    if(id===OPS.save||id===OPS.paintFormXObjectBegin){stack.push({matrix:[...matrix],width});if(id===OPS.paintFormXObjectBegin&&args[0])matrix=Util.transform(matrix,args[0]);}
    else if(id===OPS.restore||id===OPS.paintFormXObjectEnd){const saved=stack.pop();if(!saved)throw Error("native PDF graphics stack underflow");matrix=saved.matrix;width=saved.width;}
    else if(id===OPS.transform)matrix=Util.transform(matrix,args);
    else if(id===OPS.setLineWidth)width=Number(args[0]);
    else if(id===OPS.setGState){for(const state of args[0])if(state[0]==="LW")width=Number(state[1]);}
    else if(id===OPS.constructPath&&args[2]){
      const path=Array.from(args[1]?.[0]??[],Number);
      let cursor=0,start:number[]|undefined,lines=0,curves=0,closed=0;
      const points:number[][]=[],corners:number[][]=[];
      while(cursor<path.length){
        const command=path[cursor++]!;
        if(command===0){start=[path[cursor++]!,path[cursor++]!];if(!lines&&!curves){corners.length=0;corners.push([...start]);}}
        else if(command===1){start=[path[cursor++]!,path[cursor++]!];corners.push([...start]);lines++;}
        else if(command===2){
          if(!start)throw Error("native PDF curve without a starting point");
          const curve=[start,[path[cursor++]!,path[cursor++]!],[path[cursor++]!,path[cursor++]!],[path[cursor++]!,path[cursor++]!]];
          for(let sample=0;sample<=128;sample++){const t=sample/128,u=1-t;points.push([0,1].map(axis=>u**3*curve[0]![axis]!+3*u*u*t*curve[1]![axis]!+3*u*t*t*curve[2]![axis]!+t**3*curve[3]![axis]!));}
          start=curve[3]!;curves++;
        }else if(command===4)closed++;
        else{curves=-1;break;}
      }
      const kind=closed<=1&&curves===4&&lines===0?"circle":closed===1&&curves===0&&lines===3?"rect":undefined;
      if(kind==="rect"&&args[0]===OPS.stroke&&Math.abs(width-point(.01))<.002&&Array.from(args[2],Number).every((value,axis)=>Math.abs(value-[0,0,point(canvas[0]!),point(canvas[1]!)][axis]!)<point(.01))){
        if(reference)throw Error("duplicate native PDF canvas reference");reference=[...matrix];
      }else if(kind)shapes.push({kind,points:kind==="circle"?points:corners,matrix:[...matrix],paint:args[0]});
    }
  }
  if(stack.length)throw Error("unbalanced native PDF graphics stack");
  if(!reference)throw Error("native PDF canvas reference missing");
  const inverse=Util.inverseTransform(reference);
  return shapes.map(shape=>{
    const relative=Util.transform(inverse,shape.matrix);
    shape.points.forEach(value=>Util.applyTransform(value,relative));
    const bounds=[Math.min(...shape.points.map(value=>value[0]!)),Math.min(...shape.points.map(value=>value[1]!)),Math.max(...shape.points.map(value=>value[0]!)),Math.max(...shape.points.map(value=>value[1]!))].map(point.invert);
    return {kind:shape.kind,bounds,paint:shape.paint};
  });
}

/** 📐️ Reads straight closed PDF paths relative to declared per-case physical canvas markers. */
function nativeGeoPlanarPdfPaths(operators:Awaited<ReturnType<PDFPageProxy["getOperatorList"]>>,references:number[][],result:Map<number,{points:number[][];width:number;opacity:number;fillOpacity:number;join:number}[]>,markers:string[]):void{
  let matrix=[1,0,0,1,0,0],width=1,opacity=1,fillOpacity=1,lineJoin=0,reference:number[]|undefined,owner:number|undefined;
  const stack:{matrix:number[];width:number;opacity:number;fillOpacity:number;join:number}[]=[],content:string[]=[],mm=72/25.4;
  for(const[index,id]of operators.fnArray.entries()){
    const args=operators.argsArray[index];
    if(id===OPS.beginMarkedContent||id===OPS.beginMarkedContentProps)content.push(String(args?.[0]?.name??args?.[0]));
    else if(id===OPS.endMarkedContent){if(!content.length)throw Error("Planar PDF ownership stack underflow");content.pop();}
    else if(id===OPS.save||id===OPS.paintFormXObjectBegin){stack.push({matrix:[...matrix],width,opacity,fillOpacity,join:lineJoin});if(id===OPS.paintFormXObjectBegin&&args[0])matrix=Util.transform(matrix,args[0]);}
    else if(id===OPS.restore||id===OPS.paintFormXObjectEnd){const value=stack.pop();if(!value)throw Error("Planar PDF stack underflow");matrix=value.matrix;width=value.width;opacity=value.opacity;fillOpacity=value.fillOpacity;lineJoin=value.join;}
    else if(id===OPS.transform)matrix=Util.transform(matrix,args);
    else if(id===OPS.setLineWidth)width=Number(args[0]);
    else if(id===OPS.setLineJoin)lineJoin=Number(args[0]);
    else if(id===OPS.setGState){for(const item of args[0]){if(item[0]==="LW")width=Number(item[1]);if(item[0]==="CA")opacity=Number(item[1]);if(item[0]==="ca")fillOpacity=Number(item[1]);}}
    else if(id===OPS.constructPath&&args[2]){
      const packed=Array.from(args[1]?.[0]??[],Number),points:number[][]=[];let cursor=0,closed=0,valid=true;
      while(cursor<packed.length){const command=packed[cursor++]!;if(command===0||command===1)points.push([packed[cursor++]!,packed[cursor++]!]);else if(command===4)closed++;else{valid=false;break;}}
      if(!valid||closed!==1)continue;
      for(let point=points.length-1;point>0;point--)if(points[point]![0]===points[point-1]![0]&&points[point]![1]===points[point-1]![1])points.splice(point,1);
      const found=new Set(points.map(point=>point.join(","))).size===4&&Math.abs(width-.01*mm)<.002?references.findIndex(canvas=>Array.from(args[2],Number).every((value,axis)=>Math.abs(value-[0,0,canvas[0]!*mm,canvas[1]!*mm][axis]!)<.03)):-1;
      if(found>=0){owner=found;reference=[...matrix];if(result.has(owner))throw Error("Planar duplicate canvas marker");result.set(owner,[]);}
      else if(owner!==undefined&&reference&&content.includes(markers[owner]!)&&points.length>=3){const transform=Util.transform(Util.inverseTransform(reference),matrix);points.forEach(point=>Util.applyTransform(point,transform));result.get(owner)!.push({points:points.map(point=>point.map(value=>value/mm)),width:width/mm,opacity,fillOpacity,join:lineJoin});}
    }
  }
  if(stack.length)throw Error("Planar unbalanced PDF stack");
  if(content.length)throw Error("Planar unbalanced PDF ownership stack");
}
