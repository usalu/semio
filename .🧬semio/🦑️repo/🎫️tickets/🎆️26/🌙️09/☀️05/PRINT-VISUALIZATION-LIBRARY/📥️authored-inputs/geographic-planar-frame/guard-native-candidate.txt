% 🛟️ Rejects degenerate planar frames and raw kernels before any binning or drawing.
\bool_new:N \l_semio_viz_geosym_valid_bool
\fp_new:N \l_semio_viz_geosym_stroke_fp
\msg_new:nnn { semio-viz } { planar-domain }
  { Planar~display~#1~requires~finite~positive~dimensions/radius/bandwidth~and~a~nonnegative~stroke~smaller~than~both~frame~axes. }

% 🔎️ Checks one evaluated raw/frame value against its strictly positive native domain.
\cs_new_protected:Npn \semio_viz_geosym_positive:nn #1#2
  {
    \fp_if_nan:nTF {#2}
      { \semio_viz_geosym_invalid:n {#1} }
      { \fp_compare:nF { abs (#2) < inf && #2 > 0 }
          { \semio_viz_geosym_invalid:n {#1} } }
  }

% 🚫️ Marks a native domain violation without publishing invalid planar geometry.
\cs_new_protected:Npn \semio_viz_geosym_invalid:n #1
  {
    \bool_set_false:N \l_semio_viz_geosym_valid_bool
    \msg_error:nnn { semio-viz } { planar-domain } {#1}
  }

% 📏️ Evaluates physical TeX stroke units and all source-owned planar domain relations.
\cs_new_protected:Npn \semio_viz_geosym_validate:
  {
    \bool_set_true:N \l_semio_viz_geosym_valid_bool
    \semio_viz_geosym_positive:nn { width } { \l_semio_viz_geo_f_width_fp }
    \semio_viz_geosym_positive:nn { height } { \l_semio_viz_geo_f_height_fp }
    \semio_viz_geosym_positive:nn { radius } { \l_semio_viz_geosym_radius_fp }
    \semio_viz_geosym_positive:nn { bandwidth } { \l_semio_viz_geosym_bandwidth_fp }
    \fp_set:Nn \l_semio_viz_geosym_stroke_fp
      { \dim_to_fp:n { \l_semio_viz_geo_f_linewidth_tl } / \dim_to_fp:n { 1mm } }
    \fp_compare:nF
      { \l_semio_viz_geosym_stroke_fp >= 0
        && \l_semio_viz_geosym_stroke_fp < \l_semio_viz_geo_f_width_fp
        && \l_semio_viz_geosym_stroke_fp < \l_semio_viz_geo_f_height_fp }
      { \semio_viz_geosym_invalid:n { lineWidth } }
  }
