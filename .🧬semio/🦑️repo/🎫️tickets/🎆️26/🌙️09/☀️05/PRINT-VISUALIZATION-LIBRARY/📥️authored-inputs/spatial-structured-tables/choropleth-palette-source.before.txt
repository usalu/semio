%region 🔖️Classes
% 📐 A choropleth classifies part values into colour classes and reads the colour ramp at the class
% position. The class rule is a scale of the scale kernel: `quantize`, `quantile` and `threshold`
% are its binned kinds, every other rule is the continuous normalisation of the value extent. The
% range of that scale is the evenly spaced ramp positions, so a class maps straight onto [0,1].
\seq_new:N \l_semio_viz_chor_values_seq
\tl_new:N \l_semio_viz_chor_scale_tl
\clist_new:N \l_semio_viz_chor_breaks_clist
\clist_new:N \l_semio_viz_chor_stops_clist
\int_new:N \l_semio_viz_chor_classes_int
\tl_new:N \l_semio_viz_chor_pos_tl

\cs_new_protected:Npn \semio_viz_chor_collect:nnnn #1#2#3#4
  { \seq_put_right:Nn \l_semio_viz_chor_values_seq {#3} }

% 🪜 Evenly spaced ramp positions for #1 classes: 0, 1/(n-1), …, 1.
\cs_new_protected:Npn \semio_viz_chor_stops:n #1
  {
    \clist_clear:N \l_semio_viz_chor_stops_clist
    \int_step_inline:nn {#1}
      {
        \clist_put_right:Nx \l_semio_viz_chor_stops_clist
          { \fp_to_decimal:n { ( ##1 - 1 ) / max ( 1 , (#1) - 1 ) } }
      }
  }

% 🧮 Registers the class rule of geometry #1 as the scale `semio-choropleth`.
\cs_new_protected:Npn \semio_viz_chor_prepare:n #1
  {
    \semio_viz_geo_value_extent:n {#1}
    \seq_clear:N \l_semio_viz_chor_values_seq
    \semio_viz_geom_map_inline:nn {#1}
      { \semio_viz_chor_collect:nnnn ##1 }
    \str_case:VnF \l_semio_viz_chor_scale_tl
      {
        { quantile }
          {
            \int_set:Nn \l_semio_viz_chor_classes_int
              { \int_max:nn { 2 } { \clist_count:N \l_semio_viz_chor_breaks_clist + 1 } }
            \semio_viz_chor_stops:n { \l_semio_viz_chor_classes_int }
            \semio_viz_scale_define:nnxx { semio-choropleth } { quantile }
              { \seq_use:Nn \l_semio_viz_chor_values_seq { , } }
              { \clist_use:Nn \l_semio_viz_chor_stops_clist { , } }
          }
        { threshold }
          {
            \int_set:Nn \l_semio_viz_chor_classes_int
              { \clist_count:N \l_semio_viz_chor_breaks_clist + 1 }
            \semio_viz_chor_stops:n { \l_semio_viz_chor_classes_int }
            \semio_viz_scale_define:nnxx { semio-choropleth } { threshold }
              { \clist_use:Nn \l_semio_viz_chor_breaks_clist { , } }
              { \clist_use:Nn \l_semio_viz_chor_stops_clist { , } }
          }
        { quantize }
          {
            \int_set:Nn \l_semio_viz_chor_classes_int
              { \int_max:nn { 2 } { \clist_count:N \l_semio_viz_chor_breaks_clist + 1 } }
            \semio_viz_chor_stops:n { \l_semio_viz_chor_classes_int }
            \semio_viz_scale_define:nnxx { semio-choropleth } { quantize }
              { \fp_use:N \l_semio_viz_geo_vmin_fp , \fp_use:N \l_semio_viz_geo_vmax_fp }
              { \clist_use:Nn \l_semio_viz_chor_stops_clist { , } }
          }
      }
      { \tl_clear:N \l_semio_viz_chor_scale_tl }
    \semio_viz_chor_classify:
  }

% 📐 Class positions are precomputed per value so that the fill style stays expandable: a TikZ
% option list cannot run the protected scale kernel while it is being read.
\prop_new:N \l_semio_viz_chor_pos_prop

\cs_new_protected:Npn \semio_viz_chor_classify:
  {
    \prop_clear:N \l_semio_viz_chor_pos_prop
    \seq_map_inline:Nn \l_semio_viz_chor_values_seq
      { \semio_viz_chor_classify_one:n {##1} }
  }

% 🎨 Records the ramp position of one value: its class under a class rule, its own share without one.
\cs_new_protected:Npn \semio_viz_chor_classify_one:n #1
  {
    \tl_if_empty:NTF \l_semio_viz_chor_scale_tl
      { \tl_set:Nx \l_semio_viz_chor_pos_tl { \semio_viz_geo_norm:n {#1} } }
      { \semio_viz_scale_binned_item:nnN { semio-choropleth } {#1} \l_semio_viz_chor_pos_tl }
    \prop_put:NnV \l_semio_viz_chor_pos_prop {#1} \l_semio_viz_chor_pos_tl
  }

% 📐 Ramp position of a value under the active class rule; the key is expanded so a token list resolves.
\cs_generate_variant:Nn \prop_item:Nn { Ne }
\cs_new:Npn \semio_viz_chor_position:n #1
  { \prop_item:Ne \l_semio_viz_chor_pos_prop {#1} }
%endregion 🔖️Classes

%region 🔖️Keys
% 🔑 semio / viz / family / geo-choropleth — inherits semio / viz / geo / base, plus
% 🔑 scale    (choice) quantize | quantile | threshold | sequential | diverging; default quantize.
% 🔑 breaks   (clist)  explicit class breaks for scale=threshold; default 10,30,50.
% 🔑 bivariate (bool)  encode value against value2 in a two-way ramp (hatch overlay); default false.
% 🔑 legend   (bool)   draw the class swatch strip below the map; default true.
% 🔑 dasymetric (bool) shrink each polygon towards its centroid by `weight` to approximate population density; default false.
% 🔑 weight   (0..1)   dasymetric shrink factor; default 0.7.
\bool_new:N \l_semio_viz_chor_bivariate_bool
\bool_new:N \l_semio_viz_chor_legend_bool

  }

    \semio_viz_geo_base_reset:
    \tl_set:Nn \l_semio_viz_chor_scale_tl { quantize }
    \clist_set:Nn \l_semio_viz_chor_breaks_clist { 10, 30, 50 }
    \bool_set_false:N \l_semio_viz_chor_bivariate_bool
    \bool_set_true:N \l_semio_viz_chor_legend_bool
    \bool_set_false:N \l_semio_viz_chor_dasymetric_bool
    \fp_set:Nn \l_semio_viz_chor_weight_fp { 0.7 }
    \tl_set:Nn \l_semio_viz_chor_cart_tl { dorling }
    \fp_set:Nn \l_semio_viz_chor_maxsize_fp { 6 }
    \tl_set:Nn \l_semio_viz_chor_tile_tl { square }
    \int_set:Nn \l_semio_viz_chor_columns_int { 4 }
    \fp_set:Nn \l_semio_viz_chor_cell_fp { 9 }
    \bool_set_true:N \l_semio_viz_chor_labels_bool
  }

% 🎨 Choropleth family: every polygon filled by its own value under the selected class rule.
\SemioVizFamily { geo-choropleth }
  {
    \semio_viz_chor_reset:
    \keys_set:nn { semio / viz / geo / base } {#1}
    \keys_set:nn { semio / viz / family / geo-choropleth } {#1}
    \exp_args:NV \semio_viz_geo_base_frame:n \l_semio_viz_geo_f_geometry_tl
    \exp_args:NV \semio_viz_chor_prepare:n \l_semio_viz_geo_f_geometry_tl
    \bool_if:NTF \l_semio_viz_chor_dasymetric_bool
      { \semio_viz_chor_dasymetric: }
      {
        \semio_viz_geo_draw_geometry:nnn { semio-family }
          { \l_semio_viz_geo_f_geometry_tl } { \semio_viz_chor_fill: }
      }
    \bool_if:NT \l_semio_viz_chor_bivariate_bool { \semio_viz_chor_hatch: }
    \bool_if:NT \l_semio_viz_chor_legend_bool { \semio_viz_chor_legend: }
    \semio_viz_chor_probe:
  }

% 🧭 The classification as numbers: the class rule, the step count and, per part, the ramp position
% its fill came from. Without it a choropleth's probe output is the outline every geo family draws.
\cs_new_protected:Npn \semio_viz_chor_probe:
  {
    \bool_if:NT \g_semio_viz_probe_geometry_bool
      {
        \semio_viz_probe_geometry:nx { geo/choropleth/scale }
          {
            \tl_if_empty:NTF \l_semio_viz_chor_scale_tl { linear }
              { \l_semio_viz_chor_scale_tl } ,
            \int_use:N \l_semio_viz_geo_f_steps_int ,
            \bool_if:NTF \l_semio_viz_chor_bivariate_bool { 1 } { 0 }
          }
        \semio_viz_geo_map_anchors:nnn { semio-family }
          { \l_semio_viz_geo_f_geometry_tl } { \semio_viz_chor_class:nnnn }
      }
  }

% 🎨 One classified part as numbers: its centroid, its value and the ramp position it was filled with.
\cs_new_protected:Npn \semio_viz_chor_class:nnnn #1#2#3#4
  {
    \semio_viz_probe_geometry:nx { geo/choropleth/class }
      {
        \fp_to_decimal:n {#1} , \fp_to_decimal:n {#2} , \fp_to_decimal:n {#4} ,
        \fp_to_decimal:n { \semio_viz_chor_position:n {#4} }
      }
  }

% 🖌️ TikZ keys of one polygon: the ramp colour of the class its value fell into.
% 🏘️ The dasymetric case: the class colour is carried by a copy of every polygon pulled `weight` of
% the way towards its own centroid, over the full outline in hairline. The shading then covers the
% part of the polygon the value is attributed to instead of its whole administrative area.
\cs_new_protected:Npn \semio_viz_chor_dasymetric:
  {
    \semio_viz_geo_draw_geometry:nnn { semio-family }
      { \l_semio_viz_geo_f_geometry_tl }
      { draw=semio-chrome-border-normal, line~width=\semio@stroke@hairline, fill=none }
    \fp_set_eq:NN \l_semio_viz_chor_shrink_fp \l_semio_viz_chor_weight_fp
    \semio_viz_geom_new:n { semio-family-dasymetric }
    \exp_args:NV \semio_viz_geom_map_inline:nn \l_semio_viz_geo_f_geometry_tl
      { \semio_viz_chor_dasymetric_ring:nnnn ##1 }
    \semio_viz_geo_draw_geometry:nnn { semio-family }
      { semio-family-dasymetric } { \semio_viz_chor_fill: }
  }

% 🏘️ One shrunk polygon of the dasymetric copy.
\cs_new_protected:Npn \semio_viz_chor_dasymetric_ring:nnnn #1#2#3#4
  { \semio_viz_chor_shrink_ring:nnnn { semio-family-dasymetric } {#2} {#3} {#4} }

% 🎨 The TikZ option text every choropleth polygon is stroked and filled with.
\cs_new:Npn \semio_viz_chor_fill:
  {
    draw= \l_semio_viz_geo_f_stroke_tl ,
    line~width= \l_semio_viz_geo_f_linewidth_tl ,
    fill= \semio_viz_geo_ramp:n
            { \semio_viz_chor_position:n { \l_semio_viz_geo_partvalue_tl } } ,
    fill~opacity= \fp_use:N \l_semio_viz_geo_f_opacity_fp
  }

% ▨ Bivariate overlay: the second variable is encoded as a dot at the part centroid whose size grows with the value.
\cs_new_protected:Npn \semio_viz_chor_hatch:
  {
    \semio_viz_geo_map_anchors:nnn { semio-family }
      { \l_semio_viz_geo_f_geometry_tl } { \semio_viz_chor_hatch_dot:nnnn }
  }

% ▨ The bivariate overlay's dot at one part centroid, its radius carrying the second variable.
\cs_new_protected:Npn \semio_viz_chor_hatch_dot:nnnn #1#2#3#4
  {
    \fill [ semio-chrome-foreground, fill~opacity=0.55 ]
      (#1,#2) circle
        [ radius= \fp_eval:n { 0.6 + 1.6 * \semio_viz_geo_norm:n {#4} } ] ;
  }

% 🏷️ Class swatch strip: one rectangle per ramp step, drawn along the bottom edge.
\cs_new_protected:Npn \semio_viz_chor_legend:
  {
    \int_step_inline:nn { \l_semio_viz_geo_f_steps_int }
      {
        \use:x
          {
            \exp_not:N \fill
              [
                \semio_viz_geo_ramp:n
                  { ( ##1 - 1 ) / ( \l_semio_viz_geo_f_steps_int - 1 ) }
              ]
              ( \fp_eval:n { ( ##1 - 1 ) * 5 } , -5 )
              rectangle
              ( \fp_eval:n { ##1 * 5 - 0.8 } , -2.4 ) ;
          }
      }
  }

% 🫧 Cartogram family: circles, squares or scaled polygons sized by value at the part centroids.
\SemioVizFamily { geo-cartogram }
  {
    \semio_viz_chor_reset:
    \keys_set:nn { semio / viz / geo / base } {#1}
    \keys_set:nn { semio / viz / family / geo-cartogram } {#1}
    \exp_args:NV \semio_viz_geo_base_frame:n \l_semio_viz_geo_f_geometry_tl
    \exp_args:NV \semio_viz_geo_value_extent:n \l_semio_viz_geo_f_geometry_tl
    \str_case:VnF \l_semio_viz_chor_cart_tl
      {
        { non-contiguous } { \semio_viz_chor_noncontiguous: }
        { contiguous } { \semio_viz_chor_contiguous: }
      }
      {
        \semio_viz_geo_draw_geometry:nnn { semio-family }
          { \l_semio_viz_geo_f_geometry_tl }
          {
            draw=semio-chrome-border-normal,
            line~width=\semio@stroke@hairline,
            fill=semio-chrome-panel, fill~opacity=0.35
          }
        \semio_viz_geo_map_anchors:nnn { semio-family }
          { \l_semio_viz_geo_f_geometry_tl } { \semio_viz_chor_blob:nnnn }
      }
  }

% 🫧 One Dorling or Demers body at a part centroid: a circle or a square whose area is the value.
\cs_new_protected:Npn \semio_viz_chor_blob:nnnn #1#2#3#4
  {
    \str_if_eq:VnTF \l_semio_viz_chor_cart_tl { demers }
      {
        \use:x
          {
            \exp_not:N \filldraw
              [
                draw= \exp_not:V \l_semio_viz_geo_f_stroke_tl ,
                fill= \semio_viz_geo_ramp:n { \semio_viz_geo_norm:n {#4} }
              ]
              ( \fp_eval:n
                  { #1 - \l_semio_viz_chor_maxsize_fp
                        * sqrt ( \semio_viz_geo_norm:n {#4} ) } ,
                \fp_eval:n
                  { #2 - \l_semio_viz_chor_maxsize_fp
                        * sqrt ( \semio_viz_geo_norm:n {#4} ) } )
              rectangle
              ( \fp_eval:n
                  { #1 + \l_semio_viz_chor_maxsize_fp
                        * sqrt ( \semio_viz_geo_norm:n {#4} ) } ,
                \fp_eval:n
                  { #2 + \l_semio_viz_chor_maxsize_fp
                        * sqrt ( \semio_viz_geo_norm:n {#4} ) } ) ;
          }
      }
      {
        \use:x
          {
            \exp_not:N \filldraw
              [
                draw= \exp_not:V \l_semio_viz_geo_f_stroke_tl ,
                fill= \semio_viz_geo_ramp:n { \semio_viz_geo_norm:n {#4} }
              ]
              (#1,#2) circle
                [ radius= \fp_eval:n
                    { 1 + \l_semio_viz_chor_maxsize_fp
                          * sqrt ( \semio_viz_geo_norm:n {#4} ) } ] ;
          }
      }
  }

% 🔻 Non-contiguous cartogram: each ring scaled about its own centroid by sqrt(value), so area encodes the value.
\cs_new_protected:Npn \semio_viz_chor_noncontiguous:

      { \semio_viz_chor_scale_ring:nnnn ##1 }

        draw=semio-chrome-foreground, line~width=\semio@stroke@hairline,

% 🔻 Contiguous cartogram: the documented iterative approximation — the same centroid scaling applied `iterations`
% 🔻 times with a shrinking step, which converges towards but never reaches a true contiguous (Gastner–Newman) result.
\cs_new_protected:Npn \semio_viz_chor_contiguous:
  {
    \semio_viz_geom_new:n { semio-family-scaled }
    \exp_args:NV \semio_viz_geom_map_inline:nn \l_semio_viz_geo_f_geometry_tl
      { \semio_viz_chor_scale_ring:nnnn ##1 }
    \semio_viz_geo_path:nn { semio-family } { semio-family-scaled }
    \semio_viz_geo_draw:n
      {
        draw=semio-chrome-foreground, line~width=\semio@stroke@hairline,
        fill=\semio_viz_theme_color:n { 0 }, fill~opacity=0.45
      }
    \semio_viz_geo_draw_geometry:nnn { semio-family }
      { \l_semio_viz_geo_f_geometry_tl }
      {
        draw=semio-chrome-border-normal, line~width=\semio@stroke@hairline,
        dash~pattern=on~1pt~off~1pt, fill=none
      }
  }

% 🔍 One non-contiguous cartogram ring: the part's own outline scaled about its centroid by its value.
\cs_new_protected:Npn \semio_viz_chor_scale_ring:nnnn #1#2#3#4
  {
    \fp_set:Nn \l_semio_viz_chor_shrink_fp
      { 0.35 + 0.65 * sqrt ( \semio_viz_geo_norm:n {#3} ) }
    \semio_viz_chor_shrink_ring:nnnn { semio-family-scaled } {#2} {#3} {#4}
  }
\fp_new:N \l_semio_viz_chor_shrink_fp

% 🔍 Files the ring #4 of part #2, pulled towards its own centroid by `\l_semio_viz_chor_shrink_fp`,
% as a part of collection #1. Both the non-contiguous cartogram and the dasymetric choropleth are
% this one operation with a different factor.
\cs_new_protected:Npn \semio_viz_chor_shrink_ring:nnnn #1#2#3#4
  {
    \semio_viz_geo_ring_centroid:n {#4}
    \fp_set:Nn \l_tmpa_fp { \l_semio_viz_geo_lon_fp }
    \fp_set:Nn \l_tmpb_fp { \l_semio_viz_geo_lat_fp }
    \tl_clear:N \l_tmpa_tl
    \seq_set_split:Nnn \l_tmpa_seq { ; } {#4}
    \seq_map_inline:Nn \l_tmpa_seq
      {
        \tl_put_right:Nx \l_tmpa_tl
          {
            \fp_eval:n
              {
                \l_tmpa_fp + \l_semio_viz_chor_shrink_fp
                * ( \clist_item:nn {##1} { 1 } - \l_tmpa_fp )
              } ,
            \fp_eval:n
              {
                \l_tmpb_fp + \l_semio_viz_chor_shrink_fp
                * ( \clist_item:nn {##1} { 2 } - \l_tmpb_fp )
              } ;
          }
      }
    \use:x
      {
        \exp_not:N \semio_viz_geom_part:nnnn { \exp_not:n {#1} } { polygon }
          { id = #2, value = #3 } { \exp_not:V \l_tmpa_tl }
      }
  }

% ▦ Tile-grid family: one tile per part on a regular lattice, coloured by value; geography is replaced by the grid.
\SemioVizFamily { geo-tilegrid }
  {
    \semio_viz_chor_reset:
    \keys_set:nn { semio / viz / geo / base } {#1}
    \keys_set:nn { semio / viz / family / geo-tilegrid } {#1}
    \exp_args:NV \semio_viz_geo_value_extent:n \l_semio_viz_geo_f_geometry_tl
    \int_zero:N \l_tmpa_int
    \exp_args:NV \semio_viz_geom_map_inline:nn \l_semio_viz_geo_f_geometry_tl
      { \semio_viz_chor_tile:nnnn ##1 }
  }

% 🔳 One tile of the grid map: the running part index becomes a row and column of `columns` cells.
\cs_new_protected:Npn \semio_viz_chor_tile:nnnn #1#2#3#4
  {
    \fp_set:Nn \l_tmpa_fp
      {
        ( \l_tmpa_int - \l_semio_viz_chor_columns_int
          * floor ( \l_tmpa_int / \l_semio_viz_chor_columns_int ) )
        * \l_semio_viz_chor_cell_fp
      }
    \fp_set:Nn \l_tmpb_fp
      {
        - floor ( \l_tmpa_int / \l_semio_viz_chor_columns_int )
        * \l_semio_viz_chor_cell_fp
      }
    \str_case:VnF \l_semio_viz_chor_tile_tl
      {
        { hex } { \semio_viz_chor_tile_hex:nn {#2} {#3} }
        { equal-area } { \semio_viz_chor_tile_square:nnn {#2} {#3} { 1 } }
      }
      { \semio_viz_chor_tile_square:nnn {#2} {#3} { 0 } }
    \int_incr:N \l_tmpa_int
  }

% ⬜ A square tile for part #1 with value #2, its corners rounded by #3, filled from the ramp.
\cs_new_protected:Npn \semio_viz_chor_tile_square:nnn #1#2#3
  {
    \use:x
      {
        \exp_not:N \filldraw
          [
            draw= \exp_not:V \l_semio_viz_geo_f_stroke_tl ,
            line~width= \exp_not:V \l_semio_viz_geo_f_linewidth_tl ,
            fill= \semio_viz_geo_ramp:n { \semio_viz_geo_norm:n {#2} }
          ]
          ( \fp_eval:n { \l_tmpa_fp } , \fp_eval:n { \l_tmpb_fp } )
          rectangle
          ( \fp_eval:n
              { \l_tmpa_fp + \l_semio_viz_chor_cell_fp - 1
                - #3 * ( \l_semio_viz_chor_cell_fp - 1 )
                  * ( 1 - sqrt ( \semio_viz_geo_norm:n {#2} ) ) } ,
            \fp_eval:n
              { \l_tmpb_fp + \l_semio_viz_chor_cell_fp - 1
                - #3 * ( \l_semio_viz_chor_cell_fp - 1 )
                  * ( 1 - sqrt ( \semio_viz_geo_norm:n {#2} ) ) } ) ;
      }
    \semio_viz_chor_tile_label:n {#1}
  }

% ⬡ A hexagonal tile for part #1 with value #2, filled from the ramp.
\cs_new_protected:Npn \semio_viz_chor_tile_hex:nn #1#2
  {
    \use:x
      {
        \exp_not:N \filldraw
          [
            draw= \exp_not:V \l_semio_viz_geo_f_stroke_tl ,
            line~width= \exp_not:V \l_semio_viz_geo_f_linewidth_tl ,
            fill= \semio_viz_geo_ramp:n { \semio_viz_geo_norm:n {#2} }
          ]
          \clist_map_function:nN { 0, 60, 120, 180, 240, 300 }
            \semio_viz_chor_hexvertex:n
          cycle ;
      }
    \semio_viz_chor_tile_label:n {#1}
  }

% 📐 One hexagon vertex at angle #1 about the current tile's centre, ready to be joined by `--`.
\cs_new:Npn \semio_viz_chor_hexvertex:n #1
  {
    (
      \fp_eval:n
        {
          \l_tmpa_fp + \l_semio_viz_chor_cell_fp / 2
          + ( \l_semio_viz_chor_cell_fp / 2 - 0.5 ) * cosd (#1)
        } ,
      \fp_eval:n
        {
          \l_tmpb_fp + \l_semio_viz_chor_cell_fp / 2
          + ( \l_semio_viz_chor_cell_fp / 2 - 0.5 ) * sind (#1)
        }
    ) --
  }

% 🏷️ The part id #1 in the centre of the current tile, under `labels`.
\cs_new_protected:Npn \semio_viz_chor_tile_label:n #1
  {
    \bool_if:NT \l_semio_viz_chor_labels_bool
      {
        \node [ font=\c_semio_viz_theme_font_label_tl, text=semio-chrome-foreground,
                anchor=center ]
          at ( \fp_eval:n { \l_tmpa_fp + \l_semio_viz_chor_cell_fp / 2 } ,
               \fp_eval:n { \l_tmpb_fp + \l_semio_viz_chor_cell_fp / 2 } ) {#1} ;
      }
  }
