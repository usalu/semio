  const neutral=fixture.nativeHiveControls,c=neutral.vectors,validate=new Ajv2020({strict:false}).compile(neutral.schema),before=JSON.stringify(c),failures:string[]=[],cases=c.orders.flatMap(order=>c.radii.flatMap((radii,index)=>c.languages.map(language=>({id:order+"-"+index+"-"+language,order,radii,language}))));
  if(!validate(c))throw Error("Hive neutral schema "+JSON.stringify(validate.errors));
  const degree=(id:string)=>c.edges.filter(edge=>edge.source===id||edge.target===id).length,groups=[...d3Group(c.nodes,node=>node.group).keys()];
  for(const appearance of["light","dark"]as const){
    const directory=join(workDir,appearance),body:{raw:string}[]=[];
    for(const entry of cases){
      const base={chart:{language:entry.language,theme:{appearance},width:c.canvas[0],height:c.canvas[1],margin:{top:0,right:0,bottom:0,left:0},layers:[],tables:[{name:"hive-nodes",columns:["id","group"],rows:c.nodes},{name:"hive-edges",columns:["source","target","weight"],rows:c.edges}],presets:[{kind:"hive-plot",options:{nodes:"hive-nodes",data:"hive-edges",id:"id",group:"group",source:"source",target:"target",weight:"weight",order:entry.order==="index"?"degree":"index",innerRadius:entry.radii[0],outerRadius:entry.radii[1],shape:"square",labels:false,size:c.size,sizeBy:"none",padding:String(c.padding)}}]}}as unknown as VizChartSnapshot,original=JSON.stringify(base),outcome=changeVizChartValue(base,{path:["presets","0","options","order"],value:entry.order}),replay=applyVizChartDiff(base,outcome.diff),inverse=applyVizChartDiff(replay.snapshot,inverseVizChartDiff(base,outcome.diff));
      if(outcome.messages.length||outcome.diff.edits.length!==1||replay.messages.length||inverse.messages.length||JSON.stringify(base)!==original)throw Error(entry.id+" hive mutation/replay/purity refused");deepStrictEqual(inverse.snapshot,base);const inferred=await inferVizChart(replay.snapshot);if(!inferred.tikz||inferred.diagnostics.some(item=>item.code!=="print.chart.scene-unavailable"))throw Error(entry.id+" hive inference "+JSON.stringify(inferred.diagnostics));mkdirSync(join(directory,"emissions"),{recursive:true});await Bun.write(join(directory,"emissions",entry.id+".tex"),inferred.tikz);
      body.push({raw:"\\clearpage\\SemioVizProbeBegin{hive-controls}{"+entry.id+"}"+inferred.tikz.replace(/(\\begin\{VizFigure\}[^\n]*\n)/,"$1\\draw[line width=.01mm] (0,0) rectangle (80,50);\\special{pdf:literal direct /SemioVizControl"+entry.id+" BMC}\n").replace("\\end{VizFigure}","\\special{pdf:literal direct EMC}\\end{VizFigure}")});
    }
    const records=await compileVizProbeDocument({case:"hive-controls",scenario:appearance,documentClass:"semio",documentClassOptions:"type=paper,language=en,theme="+appearance,geometry:true,packages:["semio-viz"],preamble:["\\title{Hive Ordering and Radii}","\\author{Semio}","\\date{}","\\errorcontextlines=200"],body},{workDir:directory,scenario:undefined,keepWorkDir:true}),pdf=await getDocument({data:new Uint8Array(readFileSync(join(directory,"🧪️probe-out",appearance+".pdf"))),useSystemFonts:true}).promise,paints=new Map<string,ReturnType<typeof nativeControlPdfPaint>>();
    try{for(let page=1;page<=pdf.numPages;page++){const operators=await(await pdf.getPage(page)).getOperatorList();for(const entry of cases){const marker="SemioVizControl"+entry.id;if(operators.argsArray.some((args,index)=>[OPS.beginMarkedContent,OPS.beginMarkedContentProps].includes(operators.fnArray[index]!)&&String(args?.[0]?.name??args?.[0])===marker))paints.set(entry.id,nativeControlPdfPaint(operators,c.canvas,marker));}}}finally{await pdf.destroy();}await Bun.write(join(directory,"owned-paint.json"),JSON.stringify(Object.fromEntries(paints),null,2));
    for(const entry of cases){try{
      const ordered=entry.order==="name"?sort(c.nodes,node=>node.id):entry.order==="degree"?sort(c.nodes,node=>-degree(node.id)):c.nodes,points=c.nodes.map(node=>{const members=ordered.filter(value=>value.group===node.group),slot=members.findIndex(value=>value.id===node.id),radius=scaleLinear([0,Math.max(1,members.length-1)],entry.radii)(slot),angle=2*Math.PI*groups.indexOf(node.group)/groups.length;return[radius*Math.cos(angle),radius*Math.sin(angle)];}),xs=points.map(point=>point[0]!),ys=points.map(point=>point[1]!),xlo=Math.min(...xs),xhi=Math.max(...xs),ylo=Math.min(...ys),yhi=Math.max(...ys),fit=Math.min((c.canvas[0]!-2*c.padding-4.4*c.size)/(xhi-xlo),(c.canvas[1]!-2*c.padding-4.4*c.size)/(yhi-ylo)),x=scaleLinear([xlo,xhi],[(c.canvas[0]!-fit*(xhi-xlo))/2,(c.canvas[0]!+fit*(xhi-xlo))/2]),y=scaleLinear([ylo,yhi],[(c.canvas[1]!-fit*(yhi-ylo))/2,(c.canvas[1]!+fit*(yhi-ylo))/2]),expected=points.map(point=>[x(point[0]!),y(point[1]!)]),nodes=records.filter(record=>record.scenario===entry.id&&record.key==="geometry/node"),paint=paints.get(entry.id);
      if(nodes.length!==c.nodes.length||!paint)throw Error("Hive native node/marker inventory differs");const marks=paint.paths.filter(path=>path.closed&&Math.abs(path.bounds[2]!-path.bounds[0]!-2*c.size)<c.toleranceMm&&Math.abs(path.bounds[3]!-path.bounds[1]!-2*c.size)<c.toleranceMm);if(marks.length!==c.nodes.length)throw Error("Hive actual body inventory "+marks.length);
      expected.forEach((point,index)=>{equal(nodes[index]!.values.slice(0,2).map(Number),point,entry.id+" independent D3 node "+c.nodes[index]!.id,.001);const mark=marks[index]!;equal([(mark.bounds[0]!+mark.bounds[2]!)/2,(mark.bounds[1]!+mark.bounds[3]!)/2],point,entry.id+" actual PDF spoke node "+c.nodes[index]!.id,c.toleranceMm);});
    }catch(error){failures.push(appearance+"/"+entry.id+" "+String(error));}}
  }
  if(JSON.stringify(c)!==before)throw Error("Hive neutral fixture changed");if(failures.length)throw Error("Native hive control effects:\n"+failures.join("\n"));console.log("[DEBUG] Hive 24 actual ordering/radius cases and 216 node bodies matched independent D3 and canonical replay/inverse/purity");
}
