export async function compileNativeGenomeWindowControls(workDir:string):Promise<void>{
  const neutral=fixture.nativeGenomeWindowControls,c=neutral.vectors,validate=new Ajv2020({strict:false}).compile(neutral.schema),before=JSON.stringify(c),failures:string[]=[],cases=[...c.modes.flatMap(entry=>c.windows.map(window=>({...entry,...window,id:entry.mode+"-"+window.id}))),...c.extraWindows.map(window=>({...c.modes[0]!,...window,id:"gene-structure-"+window.id}))];
  if(!validate(c))throw Error("Genome neutral schema "+JSON.stringify(validate.errors));
  for(const[appearance,language]of[["light","en"],["dark","de"]]as const){
    const directory=join(workDir,appearance),body:{raw:string}[]=[];
    for(const entry of cases){
      const options={mode:entry.mode,span:c.span.join(","),features:c.features.map(feature=>[feature.start,feature.end,feature.strand,feature.kind,feature.label].join("/")).join(","),coverage:c.coverage.join(","),laneHeight:c.laneHeight,width:c.canvas[0],height:c.canvas[1],axes:false,grid:false,domain:entry.domain,range:entry.range},base={chart:{language,theme:{appearance},width:c.canvas[0],height:c.canvas[1],margin:{top:0,right:0,bottom:0,left:0},layers:[],tables:[],presets:[{kind:entry.kind,options:{...options,domain:entry.domain==="0,2000"?"0,1000":"0,2000"}}]}}as unknown as VizChartSnapshot,original=JSON.stringify(base),outcome=changeVizChartValue(base,{path:["presets","0","options","domain"],value:entry.domain}),replay=applyVizChartDiff(base,outcome.diff),inverse=applyVizChartDiff(replay.snapshot,inverseVizChartDiff(base,outcome.diff));
      if(outcome.messages.length||outcome.diff.edits.length!==1||replay.messages.length||inverse.messages.length||JSON.stringify(base)!==original)throw Error(entry.id+" genome mutation/replay/purity refused");deepStrictEqual(inverse.snapshot,base);const inferred=await inferVizChart(replay.snapshot);if(!inferred.tikz||inferred.diagnostics.some(item=>item.code!=="print.chart.scene-unavailable"))throw Error(entry.id+" genome inference "+JSON.stringify(inferred.diagnostics));mkdirSync(join(directory,"emissions"),{recursive:true});await Bun.write(join(directory,"emissions",entry.id+".tex"),inferred.tikz);body.push({raw:"\\clearpage\\SemioVizProbeBegin{genome-window-controls}{"+entry.id+"}\\ExplSyntaxOn\\semio_viz_probe_values:nn{window/control}{1}\\ExplSyntaxOff\n"+inferred.tikz.replace(/(\\begin\{VizFigure\}[^\n]*\n)/,"$1\\draw[line width=.01mm] (0,0) rectangle (80,40);\\special{pdf:literal direct /SemioVizControl"+entry.id+" BMC}\n").replace("\\end{VizFigure}","\\special{pdf:literal direct EMC}\\end{VizFigure}")});
    }
    await compileVizProbeDocument({case:"genome-window-controls",scenario:appearance,documentClass:"semio",documentClassOptions:"type=paper,language="+language+",theme="+appearance,geometry:true,packages:["semio-viz"],preamble:["\\title{Genomic Authored Windows}","\\author{Semio}","\\date{}","\\errorcontextlines=200"],body},{workDir:directory,scenario:undefined,keepWorkDir:true});const pdf=await getDocument({data:new Uint8Array(readFileSync(join(directory,"🧪️probe-out",appearance+".pdf"))),useSystemFonts:true}).promise,paints=new Map<string,ReturnType<typeof nativeControlPdfPaint>>();
    try{for(let page=1;page<=pdf.numPages;page++){const operators=await(await pdf.getPage(page)).getOperatorList();for(const entry of cases){const marker="SemioVizControl"+entry.id;if(operators.argsArray.some((args,index)=>[OPS.beginMarkedContent,OPS.beginMarkedContentProps].includes(operators.fnArray[index]!)&&String(args?.[0]?.name??args?.[0])===marker))paints.set(entry.id,nativeControlPdfPaint(operators,c.canvas,marker));}}}finally{await pdf.destroy();}await Bun.write(join(directory,"owned-paint.json"),JSON.stringify(Object.fromEntries(paints),null,2));
    for(const entry of cases){try{
      const paint=paints.get(entry.id);if(!paint)throw Error("Genome actual marker absent");const domain=entry.domain?entry.domain.split(",").map(Number):c.span,coverage=entry.mode==="coverage"||entry.mode==="multi",range=entry.range?entry.range.split(",").map(Number):[0,coverage?Math.max(...c.coverage)*1.1:1],x=scaleLinear(domain,[0,c.canvas[0]!]),y=scaleLinear(range,[0,c.canvas[1]!]),near=(a:number[],b:number[])=>a.length===b.length&&a.every((value,index)=>Math.abs(value-b[index]!)<c.toleranceMm),pointPresent=(points:number[][])=>paint.paths.some(path=>points.every(point=>path.points.some(actual=>near(actual,point))));
      if(coverage){const points=c.coverage.map((value,index)=>[x(scaleLinear([0,c.coverage.length-1],c.span)(index)),y(value)]);if(!pointPresent(points))throw Error("Coverage actual vertices differ from D3 "+JSON.stringify(points));}
      if(entry.mode==="pileup"){const marks=paint.paths.filter(path=>path.closed),width=Math.abs(x(c.span[0]!+.22*(c.span[1]!-c.span[0]!))-x(c.span[0]!));if(marks.length!==18)throw Error("Pileup actual lane inventory "+marks.length);for(const mark of marks){equal([mark.bounds[2]!-mark.bounds[0]!,mark.bounds[3]!-mark.bounds[1]!],[width,1.6],entry.id+" D3 genomic interval and physical lane",c.toleranceMm);}continue;}
      if(entry.mode==="synteny"){for(const feature of c.features){const points=[[x(feature.start),y(.86)],[x(feature.end),y(.86)],[x(domain[0]!+domain[1]!-feature.start),y(.14)],[x(domain[0]!+domain[1]!-feature.end),y(.14)]];if(!pointPresent(points))throw Error("Synteny actual vertices differ from D3 "+feature.label+JSON.stringify(points));}continue;}
      if(entry.mode==="coverage")continue;
      for(const feature of c.features){if(feature.kind==="intron")continue;if(feature.kind==="variant"){if(!pointPresent([[x(feature.start),y(.5)],[x(feature.start),y(.5)+c.laneHeight]]))throw Error("Variant actual mapped tick differs from D3");continue;}const half=c.laneHeight/(feature.kind==="domain"?3:2),expected=[Math.min(x(feature.start),x(feature.end)),y(.5)-half,Math.max(x(feature.start),x(feature.end)),y(.5)+half],mark=paint.paths.find(path=>path.closed&&near(path.bounds,expected));if(!mark)throw Error("Feature actual mapped rectangle differs from D3 "+feature.label+" expected="+JSON.stringify(expected));}
    }catch(error){failures.push(appearance+"/"+entry.id+" "+String(error));}}
  }
  if(JSON.stringify(c)!==before)throw Error("Genome neutral fixture changed");if(failures.length)throw Error("Native genome window effects:\n"+failures.join("\n"));console.log("[DEBUG] Genome 38 actual branch/window cases matched independent D3 body scales and canonical replay/inverse/purity");
}
