/** 🛑️ Exercises evaluated TeX dimensions and native kernel/frame predicates before drawing. */
export async function compileNativeGeoPlanarGuard(workDir:string):Promise<void>{
  let refused=false;
  const rejectDir=join(workDir,"zero-width");
  try{
    await compileVizProbeDocument({case:"geo-planar-domain",scenario:"evaluated-zero-width",documentClass:"semio",documentClassOptions:"type=paper,language=en,theme=light",packages:["semio-viz"],preamble:["\\title{Native Planar Domain}","\\author{Semio}","\\date{}"],body:[{raw:"\\begin{VizFigure}[width=100,height=90]\\SemioVizChart{hexbin-map}[width={1-1},height=30,radius=5,lineWidth=.3mm]\\end{VizFigure}"},{values:{key:"native/accepted",clist:"1"}}]},{workDir:rejectDir,keepWorkDir:true});
  }catch(error){const log=readFileSync(join(rejectDir,"🧪️probe-out/evaluated-zero-width.log"),"utf8");if(!String(error).includes("Tectonic compilation failed")||!log.includes("Planar display width requires"))throw error;refused=true;}
  if(!refused)throw Error("Native evaluated zero width was accepted instead of a planar-domain diagnostic");
  const body=fixture.nativeGeoPlanar.evaluatedCases.map(entry=>({raw:"\\SemioVizProbeBegin{geo-planar-domain}{"+entry.id+"}\\ExplSyntaxOn\\semio_viz_geosym_reset:\\keys_set:nn{semio/viz/geo/planar}{width=60,height=30,lineWidth=.3mm,"+entry.options+"}\\keys_set:nn{semio/viz/family/geo-hexbin}{"+entry.options+"}\\semio_viz_geosym_validate:\\semio_viz_probe_values:nx{native/valid}{\\bool_if:NTF\\l_semio_viz_geosym_valid_bool{1}{0}}\\ExplSyntaxOff"}));
  const records=await compileVizProbeDocument({case:"geo-planar-domain",scenario:"predicates",documentClass:"semio",documentClassOptions:"type=paper,language=en,theme=light",packages:["semio-viz"],preamble:["\\title{Native Planar Predicates}","\\author{Semio}","\\date{}","\\ExplSyntaxOn\\msg_redirect_name:nnn{semio-viz}{planar-domain}{warning}\\ExplSyntaxOff"],body},{workDir:join(workDir,"predicates"),scenario:undefined,keepWorkDir:true});
  for(const entry of fixture.nativeGeoPlanar.evaluatedCases)equal(records.find(record=>record.scenario===entry.id&&record.key==="native/valid")!.values.map(Number),[entry.valid?1:0],entry.id+" independent evaluated finite/domain predicate");
  console.log("[DEBUG] Native "+fixture.nativeGeoPlanar.evaluatedCases.length+" evaluated finite/positive/physical-stroke predicates and actual hard compiler rejection PASS");
}
