%region 🔖️Camera
% 🎥 Camera state of the 3D-to-2D projection, the scientific namespace's own: semio-viz-coordinate
% owns the planar coordinate systems of taxonomy 79 and carries no third axis. Every renderer below
% goes through \semio_viz_space_project:nnn.
\tl_new:N \l_semio_viz_space_projection_tl
\fp_new:N \l_semio_viz_space_azimuth_fp
\fp_new:N \l_semio_viz_space_elevation_fp
\fp_new:N \l_semio_viz_space_distance_fp
\fp_new:N \l_semio_viz_space_oblique_fp
\fp_new:N \l_semio_viz_space_obliquescale_fp
\fp_new:N \l_semio_viz_space_zoom_fp
\fp_new:N \l_semio_viz_space_px_fp
\fp_new:N \l_semio_viz_space_py_fp
\fp_new:N \l_semio_viz_space_pz_fp

% 🎥 Projects the world point (#1,#2,#3): screen coordinates land in \l_semio_viz_space_px_fp / _py_fp, the camera depth in _pz_fp.
\cs_new_protected:Npn \semio_viz_space_project:nnn #1#2#3
  {
    \str_case:VnF \l_semio_viz_space_projection_tl
      {
        { oblique }
          {
            \fp_set:Nn \l_semio_viz_space_px_fp
              { #1 + \l_semio_viz_space_obliquescale_fp * ( #3 ) * cosd ( \l_semio_viz_space_oblique_fp ) }
            \fp_set:Nn \l_semio_viz_space_py_fp
              { #2 + \l_semio_viz_space_obliquescale_fp * ( #3 ) * sind ( \l_semio_viz_space_oblique_fp ) }
            \fp_set:Nn \l_semio_viz_space_pz_fp { #3 }
          }
      }
      {
        \fp_set:Nn \l_semio_viz_space_px_fp
          { - ( #1 ) * sind ( \l_semio_viz_space_azimuth_fp ) + ( #2 ) * cosd ( \l_semio_viz_space_azimuth_fp ) }
        \fp_set:Nn \l_semio_viz_space_py_fp
          {
            - ( ( #1 ) * cosd ( \l_semio_viz_space_azimuth_fp ) + ( #2 ) * sind ( \l_semio_viz_space_azimuth_fp ) )
              * sind ( \l_semio_viz_space_elevation_fp )
            + ( #3 ) * cosd ( \l_semio_viz_space_elevation_fp )
          }
        \fp_set:Nn \l_semio_viz_space_pz_fp
          {
            ( ( #1 ) * cosd ( \l_semio_viz_space_azimuth_fp ) + ( #2 ) * sind ( \l_semio_viz_space_azimuth_fp ) )
              * cosd ( \l_semio_viz_space_elevation_fp )
            + ( #3 ) * sind ( \l_semio_viz_space_elevation_fp )
          }
        \str_if_eq:VnT \l_semio_viz_space_projection_tl { perspective }
          {
            \fp_set:Nn \l_semio_viz_sci_g_fp
              { \l_semio_viz_space_distance_fp
                / max ( 0.2 , \l_semio_viz_space_distance_fp - \l_semio_viz_space_pz_fp ) }
            \fp_set:Nn \l_semio_viz_space_px_fp { \l_semio_viz_space_px_fp * \l_semio_viz_sci_g_fp }
            \fp_set:Nn \l_semio_viz_space_py_fp { \l_semio_viz_space_py_fp * \l_semio_viz_sci_g_fp }
          }
      }
    \fp_set:Nn \l_semio_viz_space_px_fp { \l_semio_viz_space_zoom_fp * \l_semio_viz_space_px_fp }
    \fp_set:Nn \l_semio_viz_space_py_fp { \l_semio_viz_space_zoom_fp * \l_semio_viz_space_py_fp }
  }

% 🎥 Camera keys shared by every 3D family:
%    projection (orthographic|isometric|axonometric|perspective|oblique, isometric).
%    azimuth (fp, 45), elevation (fp, 35.264): the view angles in degrees; `isometric` fixes them.
%    distance (fp, 60): perspective eye distance. obliqueAngle (fp, 45), obliqueScale (fp, 0.5): cabinet parameters.
%    zoom (fp, 1): millimetres per world unit.
\cs_new_protected:Npn \semio_viz_space_keys_camera:n #1
  {
    \keys_define:nn { semio / viz / family / #1 }
      {
        projection   .tl_set:N = \l_semio_viz_space_projection_tl ,
        azimuth      .fp_set:N = \l_semio_viz_space_azimuth_fp ,
        elevation    .fp_set:N = \l_semio_viz_space_elevation_fp ,
        distance     .fp_set:N = \l_semio_viz_space_distance_fp ,
        obliqueAngle .fp_set:N = \l_semio_viz_space_oblique_fp ,
        obliqueScale .fp_set:N = \l_semio_viz_space_obliquescale_fp ,
        zoom         .fp_set:N = \l_semio_viz_space_zoom_fp ,
      }
  }
% 🎥 `isometric` and `orthographic` are camera presets, applied after the option list is read.
\cs_new_protected:Npn \semio_viz_space_camera_apply:
  {
    \str_case:VnF \l_semio_viz_space_projection_tl
      {
        { isometric }
          {
            \fp_set:Nn \l_semio_viz_space_azimuth_fp { 45 }
            \fp_set:Nn \l_semio_viz_space_elevation_fp { 35.26438968275465 }
          }
        { orthographic }
          {
            \fp_set:Nn \l_semio_viz_space_azimuth_fp { 0 }
            \fp_set:Nn \l_semio_viz_space_elevation_fp { 0 }
          }
      }
      { }
  }
%endregion 🔖️Camera

%region 🔖️PaintersAlgorithm
% 🖌️ Facet buffer: each item is `depth/style/x1,y1 x2,y2 …`; the buffer is drawn back to front so nearer facets cover farther ones.
\seq_new:N \l_semio_viz_space_facets_seq
\tl_new:N \l_semio_viz_space_poly_tl
\cs_new_protected:Npn \semio_viz_space_facet_begin:
  { \tl_clear:N \l_semio_viz_space_poly_tl \fp_zero:N \l_semio_viz_sci_e_fp \int_zero:N \l_tmpb_int }
% 📌 Projects one corner of the facet under construction, appends its screen coordinates to the polygon and accumulates the camera depth for the average.
\cs_new_protected:Npn \semio_viz_space_facet_point:nnn #1#2#3
  {
    \semio_viz_space_project:nnn {#1} {#2} {#3}
    \tl_put_right:Nx \l_semio_viz_space_poly_tl
      {
        \tl_if_empty:NF \l_semio_viz_space_poly_tl { \c_space_tl }
        \fp_to_decimal:N \l_semio_viz_space_px_fp , \fp_to_decimal:N \l_semio_viz_space_py_fp
      }
    \fp_add:Nn \l_semio_viz_sci_e_fp { \l_semio_viz_space_pz_fp }
    \int_incr:N \l_tmpb_int
  }
% 📦 Closes the facet and files it in the buffer as `mean depth / style / points`, the mean being the accumulated depth over the corner count.
\cs_new_protected:Npn \semio_viz_space_facet_end:n #1
  {
    \seq_put_right:Nx \l_semio_viz_space_facets_seq
      {
        \fp_to_decimal:n { \l_semio_viz_sci_e_fp / max ( 1 , \l_tmpb_int ) } /
        \exp_not:n {#1} / \exp_not:V \l_semio_viz_space_poly_tl
      }
  }
% 🖌️ Sorts the buffer by camera depth and strokes it; the sort is the painter's algorithm itself.
\cs_new_protected:Npn \semio_viz_space_facets_paint:
  {
    \seq_sort:Nn \l_semio_viz_space_facets_seq
      {
        \fp_compare:nNnTF
          { \semio_viz_space_depth_of:n {##1} } > { \semio_viz_space_depth_of:n {##2} }
          { \sort_return_swapped: } { \sort_return_same: }
      }
    \seq_map_inline:Nn \l_semio_viz_space_facets_seq { \semio_viz_space_facet_draw:n {##1} }
    \seq_clear:N \l_semio_viz_space_facets_seq
  }
% 🖌️ Camera depth of a buffered facet: the leading field of its `depth/style/points` record.
\cs_new:Npn \semio_viz_space_slash_first:w #1 / #2 \q_stop { #1 }
% 📏 Wraps the delimited scanner so the sort comparison can ask a whole facet record for its depth.
\cs_new:Npn \semio_viz_space_depth_of:n #1 { \semio_viz_space_slash_first:w #1 \q_stop }
% 🖍️ Unpacks one buffered facet, rebuilds its space-separated corners into a closed path, strokes it in its own style and reports its depth.
\cs_new_protected:Npn \semio_viz_space_facet_draw:n #1
  {
    \seq_set_split:Nnn \l_tmpa_seq { / } {#1}
    \tl_clear:N \l_semio_viz_sci_path_tl
    \seq_set_split:Nnx \l_tmpb_seq { ~ } { \seq_item:Nn \l_tmpa_seq { 3 } }
    \seq_map_inline:Nn \l_tmpb_seq
      {
        \tl_if_empty:NF \l_semio_viz_sci_path_tl { \tl_put_right:Nn \l_semio_viz_sci_path_tl { -- } }
        \tl_put_right:Nn \l_semio_viz_sci_path_tl { ( ##1 ) }
      }
    \use:x
      {
        \exp_not:N \draw [ \seq_item:Nn \l_tmpa_seq { 2 } ]
          \exp_not:V \l_semio_viz_sci_path_tl ~ -- ~ cycle ;
      }
    \semio_viz_sci_report_values:nn { space/facet } { \seq_item:Nn \l_tmpa_seq { 1 } }
  }
%endregion 🔖️PaintersAlgorithm

% ⚪ Marks a world point: projected, then filled as a disc whose radius grows with camera depth so nearer points read larger, and reported.
\cs_new_protected:Npn \semio_viz_space_dot:nnnn #1#2#3#4
  {
    \semio_viz_space_project:nnn {#1} {#2} {#3}
    \fill [ fill=#4 ]
      ( \fp_eval:n { \l_semio_viz_space_px_fp } , \fp_eval:n { \l_semio_viz_space_py_fp } )
      circle [ radius=\fp_eval:n { 0.6 + 0.3 * ( 1 + \l_semio_viz_space_pz_fp / max ( 1 , \l_semio_viz_space_extent_fp ) ) } ] ;
    \semio_viz_sci_report_values:nn { space/point }
      { \fp_to_decimal:N \l_semio_viz_space_px_fp , \fp_to_decimal:N \l_semio_viz_space_py_fp ,
        \fp_to_decimal:N \l_semio_viz_space_pz_fp }
  }
