\documentclass{article}
\usepackage{xcolor}
\usepackage{calc}
\definecolor{semio-chrome-border-normal}{HTML}{808080}
\definecolor{semio-chrome-text-normal}{HTML}{404040}
\definecolor{semio-chrome-canvas}{HTML}{f0ead6}
\makeatletter

\newlength{\semio@stroke@hairline}
\setlength{\semio@stroke@hairline}{0.5pt}
\newlength{\semio@chrome@titlebar@height}
\setlength{\semio@chrome@titlebar@height}{14pt}
\newlength{\semio@chrome@padding}
\setlength{\semio@chrome@padding}{4pt}
\newlength{\semio@window@gap@w}
\newlength{\semio@window@cap@w}
\newlength{\semio@window@cap@body@h}
\newsavebox{\semio@window@cap@slot}
\newsavebox{\semio@window@tab@box}
\newsavebox{\semio@window@ctrl@box}

\newcommand{\semio@heading@chip@font}{\normalfont\small}

\newcommand{\semio@window@cap@body@h@set}{%
  \setlength{\semio@window@cap@body@h}{%
    \dimexpr\semio@chrome@titlebar@height-\semio@stroke@hairline\relax
  }%
}
\newcommand{\semio@window@stroke@h}[2]{%
  \color{#1}%
  \rule{\dimexpr#2\relax}{\semio@stroke@hairline}%
}
\newcommand{\semio@window@stroke@v}[1]{%
  \color{#1}%
  \rule{\semio@stroke@hairline}{\semio@chrome@titlebar@height}%
}
\newcommand{\semio@window@cap@raise}[1]{%
  \raisebox{0pt}[\dimexpr\semio@chrome@titlebar@height\relax][0pt]{#1}%
}
\newcommand{\semio@window@cap@paint}[3]{%
  \hbox to #1{%
    \rlap{%
      \color{#2}%
      \rule{#1}{#3}%
    }%
    \vbox to #3{%
      \vfil
      \nointerlineskip
      \hbox to #1{\hfil\usebox{\semio@window@cap@slot}\hfil}%
      \vfil
    }%
  }%
}
\newcommand{\semio@window@cap@muted@vbox}[1]{%
  \vbox to \semio@chrome@titlebar@height{%
    \nointerlineskip
    \hbox{\semio@window@stroke@h{semio-chrome-border-normal}{\semio@window@cap@w}}%
    \nointerlineskip
    \hbox{\semio@window@cap@paint{\semio@window@cap@w}{#1}{\semio@window@cap@body@h}}%
  }%
}
\newcommand{\semio@heading@cap@muted@core}[1]{%
  \semio@window@cap@body@h@set
  \sbox{\semio@window@cap@slot}{%
    \begingroup
    \semio@heading@chip@font\color{semio-chrome-text-normal}%
    \hspace{\semio@chrome@padding}#1\hspace{\semio@chrome@padding}%
    \endgroup
  }%
  \setlength{\semio@window@cap@w}{\wd\semio@window@cap@slot}%
  \semio@window@cap@raise{\semio@window@cap@muted@vbox{semio-chrome-canvas}}%
}
\newcommand{\semio@heading@cap@muted@tab}[1]{%
  \hbox{%
    \semio@heading@cap@muted@core{#1}%
    \semio@window@cap@raise{\semio@window@stroke@v{semio-chrome-border-normal}}%
  }%
}
\newcommand{\semio@heading@cap@muted}[1]{%
  \hbox{%
    \semio@window@cap@raise{\semio@window@stroke@v{semio-chrome-border-normal}}%
    \semio@heading@cap@muted@core{#1}%
    \semio@window@cap@raise{\semio@window@stroke@v{semio-chrome-border-normal}}%
  }%
}

\newcommand{\semio@window@header@gap@paint}[1]{%
  \semio@window@cap@body@h@set
  \semio@window@cap@raise{%
    \vbox to \semio@chrome@titlebar@height{%
      \hbox to \dimexpr\semio@window@gap@w\relax{\rule{0pt}{\semio@window@cap@body@h}}%
      \nointerlineskip
      \hbox to \dimexpr\semio@window@gap@w\relax{%
        \semio@window@stroke@h{#1}{\semio@window@gap@w}%
      }%
    }%
  }%
}

\newcommand{\semio@window@header@row@wrap}[1]{%
  \ifvmode\else\par\fi
  \noindent
  #1%
  \ifvmode\nointerlineskip\fi
}

\newcommand{\semio@window@header@muted}[2]{%
  \sbox{\semio@window@tab@box}{\semio@heading@cap@muted{#2}}%
  \edef\semio@window@header@numval{#1}%
  \if\relax\detokenize\expandafter{\semio@window@header@numval}\relax
    \setlength{\semio@window@gap@w}{%
      \dimexpr\linewidth-\wd\semio@window@tab@box\relax
    }%
    \semio@window@header@row@wrap{%
      \hbox to \linewidth{%
        \usebox{\semio@window@tab@box}%
        \semio@window@header@gap@paint{semio-chrome-border-normal}%
      }%
    }%
  \else
    \sbox{\semio@window@ctrl@box}{\semio@heading@cap@muted@tab{#1}}%
    \setlength{\semio@window@gap@w}{%
      \dimexpr\linewidth-\wd\semio@window@tab@box-\wd\semio@window@ctrl@box\relax
    }%
    \semio@window@header@row@wrap{%
      \hbox to \linewidth{%
        \usebox{\semio@window@tab@box}%
        \semio@window@header@gap@paint{semio-chrome-border-normal}%
        \usebox{\semio@window@ctrl@box}%
      }%
    }%
  \fi
}

\end{document}
