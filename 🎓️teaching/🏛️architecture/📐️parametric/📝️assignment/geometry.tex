    \pgfmathsetmacro{\AssignmentCylinderTangent}{atan(.45*#2/#1)}
    \draw plot[domain=\AssignmentCylinderTangent+180:\AssignmentCylinderTangent+360,samples=33,smooth]
      ({#1/2+#2*.225+#1/2*cos(\x)+#2*.225*sin(\x)},{#2*.175+#2*.175*sin(\x)});
    \draw[densely dashed] plot[domain=\AssignmentCylinderTangent:\AssignmentCylinderTangent+180,samples=33,smooth]
      ({#1/2+#2*.225+#1/2*cos(\x)+#2*.225*sin(\x)},{#2*.175+#2*.175*sin(\x)});
    \foreach \side in {0,180}{
      \draw ({#1/2+#2*.225+#1/2*cos(\AssignmentCylinderTangent+\side)+#2*.225*sin(\AssignmentCylinderTangent+\side)},{#2*.175+#2*.175*sin(\AssignmentCylinderTangent+\side)})
