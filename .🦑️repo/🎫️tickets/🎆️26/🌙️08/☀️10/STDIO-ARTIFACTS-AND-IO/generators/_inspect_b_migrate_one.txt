    art = art_path(emoji)
    row = owner_row(emoji)
    formats = row["stdio_artifacts"]

    scaffold_builder(cfg, art)
    scaffold_decomposer(cfg, art)

    write_io_register(cfg, art, formats)

    print(f"  old facets: {errs}")
