9216|  for (const entry of request.panels ?? []) {

9229|    probes.push({ kind: SECTION_KIND_TOOLS, key: "tools", hash: hashHexToWire(request.tools.hash) });

9234|    targets.push({ kind: "labels" });
9235|  }
